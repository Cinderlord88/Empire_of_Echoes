/**
 * EMPIRE OF ECHOES — OUTPUT MODIFIER
 *
 * Purpose: Read AI-generated navigation assessment markers, validate every
 *   transition against the current graph, remove hidden markers from story,
 *   then show reliable player-facing travel and exit commands.
 * Event-driven travel: show fixed phase-appropriate commands in state.message.
 *   Free roleplay never advances time; arrival is chosen by INPUT only.

 * Safety: An AI marker is NEVER accepted unless INPUT already recorded a
 *   genuine movement-intent candidate from the player's latest action.

/**
 * NAV-07: Observational prose for a standalone #EXITS request.
 * No AI-generated movement is admitted: descriptions are composed only from
 * already generated passages and already visited destinations. No new rolls.
 * This is intentionally separate from the numbered, authoritative exit menu.
 */
const eoeDescribeObservedExits = (site, exits) => {
  if (!exits.length) return "No passage out of this chamber is visible.";
  const room = getSiteNode(site, site.currentAreaId);
  const roomTitle = room ? getGraphRoomTitle(room) : "";
  const place = roomTitle ? `the ${roomTitle.toLowerCase()}` : `Area ${site.currentAreaId}`;
  const lead = exits.length === 1
    ? `From ${place}, one route is visible.`
    : `From ${place}, ${exits.length} distinct routes are visible.`;
  const openers = ["In one direction, ", "Elsewhere, ", "Another way out is apparent: ", "Finally, "];
  const prose = exits.map((exit, i) => {
    const original = String(exit.description || "A passage leads away from the room.").trim();
    const sentence = original.charAt(0).toLowerCase() + original.slice(1);
    let description = `${openers[i % openers.length]}${sentence}`;
    if (!/[.!?]$/.test(description)) description += ".";
    const dest = getSiteNode(site, exit.to);
    if (Number(exit.to) === 0) {
      description += " That hatch leads back aboard the Eirseid ship.";
    } else if (dest && dest.visited) {
      const knownTitle = getGraphRoomTitle(dest);
      description += ` You recognize this route toward Area ${dest.areaId}${knownTitle ? `, the ${knownTitle.toLowerCase()}` : ""}.`;
    }
    if (exit.status && (exit.status.locked || exit.status.blocked))
      description += " The passage is currently obstructed.";
    return description;
  });
  return `${lead}\n\n${prose.join(" ")}`;
};

    text = report + (aiProse?`\n\n${aiProse}`:"");

  // ========================================================================

      state.message = site
        ? `ARRIVAL COMPLETE\nSite: ${site.name}\n` +
          `The crew is aboard Node 0. Type #ENTER to enter the Site.`

      state.travel = { phase: "idle", direction: "outbound", currentDay: 0, tripLength: 0,

        siteGenerated: false, dayNeedsNarration: false, eventRoleplayActions: 0,

      log(`OUT HOME COMPLETE | from=${state.lastCompletedSiteId || "none"}`);
    } else if (travel.phase === "fatal") {
      state.message = "JOURNEY ENDED — fatal arrival.";
    } else if (travel.phase === "no_site") {
      state.message = "No Site discovered. Another #LIFTOFF may begin a new trip.";
    }
    if (travel.dayNeedsNarration) travel.dayNeedsNarration = false;
    // One-turn narration instruction; the rolled resolution stays in state.
    if (travel.autoResolutionPendingNarration) travel.autoResolutionPendingNarration = false;
    travel.dayPromptDue = false;
  }

  if (site && site.nodes && site.mode === "AT_SHIP" && travel && travel.phase === "exploring") {
    state.message = state.message || (site.lastMove && site.lastMove.to === 0
      ? "ABOARD EIRSEID SHIP — Node 0. Type #ENTER to return to Area 1, or #DEPART to start the return voyage."
      : "AT SITE — Aboard Eirseid ship, Node 0. Type #ENTER to enter Area 1.");
    site.navWarning = null;
  }
  if (site && site.nodes && site.mode === "EXPLORING_ACTIVE") {
    const exits = getSiteExits(site);
    // On any room arrival, guarantee observable passage descriptions even if
    // the storyteller ignores its explicit CONTEXT instruction. Only add
    // missing landmarks; avoid re-narrating passages already established by AI.
    if ((moveInThisAction || (site.exitsLookOnly && !pureExitQuery)) && exits.length) {
      const sceneLower = text.toLowerCase();
      const missing = exits.filter(exit =>
        !exit.landmark || !sceneLower.includes(String(exit.landmark).toLowerCase()));
      if (missing.length) {
        const descriptions = missing.map(exit => String(exit.description || "").trim())
          .filter(Boolean).map(sentence => /[.!?]$/.test(sentence) ? sentence : `${sentence}.`);
        if (descriptions.length) {
          text = text ? `${text}\n\nVisible from this chamber: ${descriptions.join(" ")}`
            : `Visible from this chamber: ${descriptions.join(" ")}`;
          log(`OUT PASSAGE SCENE FALLBACK | room=${site.currentAreaId} missing=${descriptions.length}/${exits.length}`);
        }

      (moveInThisAction || (site.showExits && site.lastMove &&
        Number(site.lastMove.to) === Number(site.currentAreaId))));

      // the particular exit by its persisted passage ID.

        // Manual #EXITS retains the familiar story-then-menu order.

          text = text ? `${text}\n\n${exitMenu}` : exitMenu;
        log(`OUT EXITS INLINE | room=${site.currentAreaId} auto=${autoExits} count=${exits.length} options=${exits.map(e => `${e.number}->${e.to}`).join(",")}`);
      } else if (ATDW_SETTINGS.SHOW_ROOM_EXPLORATION_POPUPS || site.navWarning) {
        // Automatic room-entry popups are optional; warnings remain visible
        // even when the setting is off, so invalid commands can be corrected.
        state.message = `AREA ${site.currentAreaId} — EXITS\n${lines.join("\n")}\n` +
          `Type #EXPLORE1, #EXPLORE2, etc., or describe where you go.\n` +
          `Type #EXITS to show exits again.`;
        if (site.navWarning) state.message = `${site.navWarning}\n\n${state.message}`;
        log(`OUT EXITS | room=${site.currentAreaId} count=${exits.length} options=${exits.map(e => `${e.number}->${e.to}`).join(",")}`);
      } else {
        log(`OUT EXITS POPUP SUPPRESSED | room=${site.currentAreaId} count=${exits.length}`);
      }
    }
    site.showExits = false;
    site.exitsLookOnly = false; // explicit #EXITS narratorial instruction is one response only
    site.exitsStandalone = false; // do not leak query-only behavior into later turns
    site.navWarning = null;
    log(`OUT GRAPH | mode=${site.mode} room=${site.currentAreaId} ` +
      `visited=${Object.values(site.nodes).filter(n => n.visited).length}/${site.totalAreas}`);
  }
  // Persistent location heading: JS state, not the model's account of
  // where the party is. Applies to EVERY exploring response, including Say,
  // Continue, combat, movement and #EXITS. Legacy rooms receive AREA only.
  if (site && site.nodes && site.mode === "EXPLORING_ACTIVE") {
    if (ATDW_SETTINGS.SHOW_AREA_HEADINGS && !pureExitQuery) {
      const currentNode = getSiteNode(site, site.currentAreaId);
      const title = getGraphRoomTitle(currentNode);
      const heading = `AREA ${site.currentAreaId}${title ? ` — ${title}` : ""}`;
      if (!text.startsWith(heading + "\n")) text = `${heading}\n\n${text}`;
      log(`OUT AREA HEADING | room=${site.currentAreaId} title=${title || "none"}`);
    } else {
      log(`OUT AREA HEADING SKIPPED | room=${site.currentAreaId} pureQuery=${pureExitQuery}`);
    }
  }
  // HP-01C5: deterministic invitation on first discovery, not a trap attack.
  if (site && site.nodes && site.mode==="EXPLORING_ACTIVE" &&
      site.lastMove && site.lastMove.actionCount===currentAction &&
      site.lastMove.firstVisit &&
      getSiteNode(site,site.currentAreaId)?.eoeMechanicalTrap?.status==="pending") {
    text=`SITE HAZARD — MECHANICAL TRAP\n`+
      `The mechanism threatens the party but has NOT struck. `+
      `Choose #TRAP EVADE (DEX), #TRAP DISARM (Technology), or leave by an existing exit.\n\n`+text;
  }
  // HP-01B3: Only health changes committed in THIS action get a report.
  // Keep the AI's scene prose intact; never create wounds from narration.
  const healthReport=eoeHealthReportForAction(currentAction);
  if (healthReport) {
    text=healthReport+(text?`\n\n${text}`:"");
    log(`OUT HEALTH EVENT | action=${currentAction} count=${eoeHealthChangesForAction(currentAction).length}`);
  }
  return { text };
};
modifier(text);
