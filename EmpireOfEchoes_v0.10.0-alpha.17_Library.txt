// ============================================================================
// VER-01: unified script release label; independent persistence metadata.
// This adds metadata to old saves without migrating or resetting game state.
// The health registry has its OWN schema (EOE_HEALTH_SCHEMA_VERSION), unchanged.
// ============================================================================
const EOE_RELEASE_VERSION = "0.10.0-alpha.17";
const EOE_RELEASE_ID = "CBT-01D6";
const EOE_SAVE_SCHEMA_VERSION = 1;
const eoeEnsureReleaseMeta = () => {
  const old = state.eoeReleaseMeta;
  if (old !== undefined && (!old || typeof old !== "object" ||
      old.saveSchemaVersion !== EOE_SAVE_SCHEMA_VERSION)) {
    log("EOE VERSION | unsupported save metadata schema; preserved unchanged");
    return false;
  }
  if (!old) {
    state.eoeReleaseMeta = {saveSchemaVersion:EOE_SAVE_SCHEMA_VERSION,
      lastInstalledVersion:EOE_RELEASE_VERSION};
    log(`EOE VERSION | release=${EOE_RELEASE_VERSION} saveSchema=${EOE_SAVE_SCHEMA_VERSION} feature=${EOE_RELEASE_ID}`);
  } else if (old.lastInstalledVersion !== EOE_RELEASE_VERSION) {
    old.lastInstalledVersion = EOE_RELEASE_VERSION;
    log(`EOE VERSION | release=${EOE_RELEASE_VERSION} saveSchema=${EOE_SAVE_SCHEMA_VERSION} feature=${EOE_RELEASE_ID}`);
  }
  return true;
};

// ============================================================================
// HP-01A–HP-01B3 — PERSISTENT HEALTH, WOUNDS AND PRESENTATION
// ATDW humans (PC + simplified NPC crew) start with 3 Wounds of capacity.
// HP-01B1 adds a safe authoritative Wound API: cumulative -2 penalty for
// nonfatal Wounds and immediate death at zero remaining capacity.
// Knockdown and automatic healing remain deferred. HP-01C2 records
// descriptive ATDW injury rolls (specific extra effects deferred). HP-01B3
// projects established Wounds into character cards and narrative cues.
// Natural Site creatures remain unrated pending their combat integration.
// CBT-01D2 rates location-scoped debug creatures once, persisted in health v1.
//
// IDs are based on party slot and Site/area, NEVER solely on a creature name.
// Stored records are authoritative; registration only fills ABSENT keys.
// Existing data (including an unknown future schema) must not be overwritten.
// The health registry never creates Story Cards or reveals undiscovered areas.
// ============================================================================
const EOE_HEALTH_SCHEMA_VERSION = 1;
const eoeHealthStore = () => {
  if (state.eoeHealth === undefined) {
    state.eoeHealth = {schemaVersion:EOE_HEALTH_SCHEMA_VERSION,records:{}};
  }
  const store = state.eoeHealth;
  if (!store || store.schemaVersion !== EOE_HEALTH_SCHEMA_VERSION ||
      !store.records || typeof store.records !== "object" ||
      Array.isArray(store.records)) {
    log("HEALTH REGISTRY | unsupported version or invalid storage; preserved unchanged");
    return null;
  }
  return store;
};
const eoeHealthRecord = id => {
  const store = eoeHealthStore();
  return store && Object.prototype.hasOwnProperty.call(store.records,id)
    ? store.records[id] : null;
};
const eoeHealthRegister = (id,kind,name,capacity) => {
  const store = eoeHealthStore();
  if (!store || typeof id !== "string" || !id ||
      !["PC","NPC","CREATURE"].includes(kind) ||
      !(capacity === null || (Number.isInteger(capacity) && capacity > 0))) return null;
  if (Object.prototype.hasOwnProperty.call(store.records,id))
    return store.records[id]; // never heal, resurrect, rename, or migrate here
  const record = {
    id, kind, name:String(name || "").trim().slice(0,120),
    maxWounds:capacity, woundsRemaining:capacity,
    injuries:[], conditions:[], status:"alive", schemaVersion:EOE_HEALTH_SCHEMA_VERSION
  };
  store.records[id] = record;
  log(`HEALTH REGISTER | id=${id} kind=${kind} capacity=${capacity===null ? "unrated" : capacity}`);
  return record;
};
const eoeEnsureCharacterHealth = () => {
  // Stable PC identity is independent of completing the skill-selection menu;
  // established saves without a numeric PC profile also receive health.
  eoeHealthRegister("pc","PC","Player",3);
  const profiles=state.eoeCrewProfiles;
  if (!profiles || typeof profiles !== "object") return;
  for (const slot of [2,3]) {
    const p=profiles[slot];
    if (p && p.kind === "NPC" && p.slot === slot &&
        typeof p.name === "string" && p.name.trim())
      eoeHealthRegister(`crew:${slot}`,"NPC",p.name,3);
  }
};
// CBT-01C: Single table for all five player-selected combat difficulties.
// Attack modifiers remain unchanged; CBT-01D2 also uses the chosen difficulty
// to bias one-time creature Wound capacity, without rerating existing enemies.
const EOE_COMBAT_DIFFICULTIES = Object.freeze({
  1:Object.freeze({name:"Easy",maxWounds:5,enemyAttack:-4}),
  2:Object.freeze({name:"Lightweight",maxWounds:4,enemyAttack:-2}),
  3:Object.freeze({name:"Average",maxWounds:3,enemyAttack:0}),
  4:Object.freeze({name:"Hard",maxWounds:3,enemyAttack:1}),
  5:Object.freeze({name:"Brutal",maxWounds:3,enemyAttack:2})
});
const eoeCombatConfig = () => {
  const settings=state.atdwSettings;
  if (!settings || typeof settings!=="object" || Array.isArray(settings)) return null;
  const level=settings.combatDifficulty;
  return Number.isSafeInteger(level) && EOE_COMBAT_DIFFICULTIES[level] || null;
};
const eoeCombatDifficultyStatus = () => {
  const cfg=eoeCombatConfig();
  if (!cfg) return {ok:false,error:"Combat difficulty settings invalid; no changes made."};
  const pc=eoeHealthRecord("pc");
  if (!eoeHealthValidWoundRecord(pc) || pc.kind!=="PC")
    return {ok:false,error:"Player health record unavailable; difficulty not changed."};
  return {ok:true,message:`COMBAT DIFFICULTY — ${state.atdwSettings.combatDifficulty} ${cfg.name}; `+
    `maximum Wounds ${pc.maxWounds}; ${pc.woundsRemaining}/${pc.maxWounds} remaining; `+
    `enemy attack modifier ${cfg.enemyAttack>=0?"+":""}${cfg.enemyAttack}. `+
    "Debug-creature autonomy is opt-in via #DEBUG CREATURE AUTO ON; natural Site autonomy is not yet integrated."};
};
const eoeCombatSetDifficulty = level => {
  const cfg=EOE_COMBAT_DIFFICULTIES[level];
  if (!cfg || !Number.isSafeInteger(level))
    return {ok:false,error:"Use #COMBAT DIFFICULTY 1 through 5."};
  const old=eoeCombatConfig();
  if (!old) return {ok:false,error:"Stored difficulty setting invalid; no changes made."};
  const pc=eoeHealthRecord("pc");
  if (!eoeHealthValidWoundRecord(pc) || pc.kind!=="PC")
    return {ok:false,error:"Player health record unavailable; no changes made."};
  if (pc.status==="dead") return {ok:false,error:"Player death is permanent; difficulty cannot revive the character."};
  const suffered=pc.maxWounds-pc.woundsRemaining;
  // Refuse an unsafe reduction rather than silently killing or erasing an injury.
  if (cfg.maxWounds<=suffered)
    return {ok:false,error:`Difficulty ${level} ${cfg.name} would leave no Wounds `+
      `(${suffered} already suffered). Heal first; no changes made.`};
  // Atomic: validation above must complete before changing either record.
  state.atdwSettings.combatDifficulty=level;
  pc.maxWounds=cfg.maxWounds;
  pc.woundsRemaining=cfg.maxWounds-suffered;
  pc.status=suffered ? "injured" : "alive";
  log(`COMBAT DIFFICULTY | level=${level} max=${cfg.maxWounds} wounded=${suffered} attack=${cfg.enemyAttack}`);
  return eoeCombatDifficultyStatus();
};
const eoeEnsureCombatDifficulty = () => {
  const settings=state.atdwSettings;
  if (!settings || typeof settings!=="object" || Array.isArray(settings)) return false;
  if (settings.combatDifficulty === undefined) settings.combatDifficulty=3;
  const cfg=eoeCombatConfig();
  if (!cfg) { log("COMBAT DIFFICULTY | unsupported setting; no migration"); return false; }
  const pc=eoeHealthRecord("pc");
  if (!pc || !eoeHealthValidWoundRecord(pc) || pc.kind!=="PC" || pc.status==="dead") return false;
  if (pc.maxWounds!==cfg.maxWounds) {
    const suffered=pc.maxWounds-pc.woundsRemaining;
    if (suffered>=cfg.maxWounds) {
      log("COMBAT DIFFICULTY | capacity mismatch is unsafe; preserved existing health");
      return false;
    }
    pc.maxWounds=cfg.maxWounds;
    pc.woundsRemaining=cfg.maxWounds-suffered;
    pc.status=suffered ? "injured" : "alive";
  }
  return true;
};

const eoeEnsureSiteCreatureHealth = site => {
  if (!site || typeof site.siteId !== "string" ||
      !/^site_[0-9]+$/.test(site.siteId) || !site.nodes ||
      typeof site.nodes !== "object") return;
  for (const node of Object.values(site.nodes)) {
    if (!node || !Number.isSafeInteger(node.areaId) || node.areaId < 1 ||
        !node.encounterCreature ||
        typeof node.encounterCreature.creatureName !== "string" ||
        !node.encounterCreature.creatureName.trim()) continue;
    eoeHealthRegister(`creature:${site.siteId}:${node.areaId}`,
      "CREATURE",node.encounterCreature.creatureName,null);
  }
};
const eoeEnsureAllHealthRecords = () => {
  if (!eoeHealthStore()) return false;
  eoeEnsureCharacterHealth();
  // Idempotently upgrade both current and archived sites without rerolling
  // encounters, opening creature cards, or exposing undiscovered locations.
  if (state.sites && typeof state.sites === "object")
    for (const site of Object.values(state.sites)) eoeEnsureSiteCreatureHealth(site);
  return true;
};


// ============================================================================
// HP-01B1 — AUTHORITATIVE WOUND MUTATION (HP-01C4 wires one travel event)
//
// Based on ATDW pp. 74–75: a nonfatal Wound gives a cumulative -2 penalty
// to actions; an exhausted human Wound capacity is fatal. No generic HP or
// damage-to-Wound conversion is invented here (combat/hazards do that later).
//
// eoeHealthApplyWounds({id, wounds, cause?, eventId?}) is an INTERNAL engine
// API, not a player command. The eventId, if supplied, identifies one hazard
// or attack resolution; a repeated eventId on the same actor is idempotent.
// Calls without eventId are allowed for test/dev use but cannot be deduped.
// The optional woundEvents ledger is additive to health schema v1, so saves
// created by HP-01A need NO migration. Invalid/unknown schemas fail closed.
// ============================================================================
const eoeHealthValidWoundRecord = r => !!(
  r && typeof r === "object" && !Array.isArray(r) &&
  r.schemaVersion === EOE_HEALTH_SCHEMA_VERSION &&
  ["PC","NPC","CREATURE"].includes(r.kind) &&
  Number.isSafeInteger(r.maxWounds) && r.maxWounds > 0 &&
  Number.isSafeInteger(r.woundsRemaining) && r.woundsRemaining >= 0 &&
  r.woundsRemaining <= r.maxWounds &&
  ["alive","injured","dead"].includes(r.status) &&
  Array.isArray(r.injuries) && Array.isArray(r.conditions) &&
  (r.woundEvents === undefined || Array.isArray(r.woundEvents))
);
// ============================================================================
// HP-01C2 — SIMPLIFIED ATDW PERSISTENT INJURY IDENTITY
// ATDW p.76–78 uses d20 injury outcomes and a second d10 table for criticals.
// Preserve the identity of each nonfatal Wound's injury, but do NOT yet apply
// the specialized -5 / Stamina / equipment / infection / permanent critical
// effects. Those need dedicated validated rules in HP-01C3/CBT-01; only the
// established -2/Wound general penalty is active in this release.
// Result 1 is recorded as critical trauma, NOT a silently invented death,
// disability or automatic recovery. C1's terminal 0-Wound rule still governs.
// Labels below are original short paraphrases of tabletop injury categories.
// ============================================================================
const EOE_INJURY_LABELS = Object.freeze([
  "Critical trauma", "Deep torso cut", "Injured arm", "Injured leg",
  "Contaminated wound", "Rib fracture", "Leg laceration", "Head impact",
  "Vision impairment", "Head trauma", "Crushed fingers", "Surface abrasion",
  "Head bruising", "Ear injury", "Upper-arm laceration", "Brow cut",
  "Nerve trauma", "Blood loss", "Severe blow", "Damaged protective clothing"
]);
const eoeHealthNewInjuries = (record, count, eventId, cause) => {
  const newEntries=[];
  for (let i=0; i<count; i++) {
    const roll=1+Math.floor(Math.random()*20);
    const existingEventCount=Array.isArray(record.woundEvents) ? record.woundEvents.length : 0;
    const entry={
      id:`inj:${record.id}:${existingEventCount}:${i+1}`,
      source:"ATDW-ADAPTED-D20", roll,
      label:EOE_INJURY_LABELS[roll-1], severity:roll===1?"critical":"ordinary",
      cause:typeof cause==="string"?cause.slice(0,140):"unspecified",
      woundEventId:eventId===undefined ? null:eventId,
      active:true, effectsDeferred:true
    };
    record.injuries.push(entry);
    newEntries.push(entry);
  }
  return newEntries;
};
const eoeHealthActiveInjuryLabels = record => {
  if (!record || !Array.isArray(record.injuries)) return [];
  return record.injuries.filter(i => i && i.active !== false &&
    typeof i.label === "string" && i.label.length <= 100)
    .map(i=>i.label + (i.severity === "critical" ? " (critical)" : ""));
};
const eoeHealthWoundPenalty = id => {
  const r = typeof id === "string" && id ? eoeHealthRecord(id) : null;
  return eoeHealthValidWoundRecord(r)
    ? (r.woundsRemaining === r.maxWounds ? 0 : -2 * (r.maxWounds - r.woundsRemaining)) : null;
};
const eoeHealthApplyWounds = request => {
  const error = reason => ({ok:false,error:reason});
  if (!request || typeof request !== "object" || Array.isArray(request))
    return error("invalid wound request");
  const {id, wounds, cause, eventId} = request;
  if (typeof id !== "string" || !id.trim()) return error("invalid character id");
  if (!Number.isSafeInteger(wounds) || wounds <= 0) return error("invalid Wound count");
  if (cause !== undefined && (typeof cause !== "string" || cause.length > 240))
    return error("invalid wound cause");
  if (eventId !== undefined &&
      (typeof eventId !== "string" || !eventId.trim() || eventId.length > 160))
    return error("invalid event identity");
  const record = eoeHealthRecord(id);
  if (!record) return error("unknown character or unsupported health store");
  if (!eoeHealthValidWoundRecord(record))
    return error("unrated or invalid health record; Wounds not changed");
  const history = record.woundEvents || [];
  if (eventId !== undefined && history.some(e => e && e.eventId === eventId)) {
    log(`HEALTH WOUND DUPLICATE | id=${id} eventId=${eventId}`);
    return {ok:true,duplicate:true,applied:0,remaining:record.woundsRemaining,
      status:record.status,fatal:record.status === "dead",
      actionPenalty:eoeHealthWoundPenalty(id)};
  }
  if (record.status === "dead" || record.woundsRemaining === 0)
    return error("character is already dead; cannot receive additional Wounds");
  const applied = Math.min(wounds,record.woundsRemaining);
  const remaining = record.woundsRemaining - applied;
  const fatal = remaining === 0;
  // Mutate state only after completing every validation and duplicate check.
  record.woundsRemaining = remaining;
  record.status = fatal ? "dead" : "injured";
  if (!record.woundEvents) record.woundEvents=[];
  record.woundEvents.push({
    eventId:eventId===undefined ? null : eventId,
    cause:cause===undefined ? "unspecified" : cause,
    requested:wounds,applied,remainingAfter:remaining,fatal
  });
  // A nonfatal Wound receives a descriptive injury. When an event inflicts
  // the final, fatal Wound, that last Wound gets no injury roll. Wounds
  // recorded by older releases are never backfilled with random injuries.
  // Human injury d20 labels should not be rolled for unknown alien anatomy.
  const nonfatalCount=record.kind==="CREATURE"?0:
    Math.min(applied, Math.max(0,remaining+applied-1));
  const injuries=eoeHealthNewInjuries(record,nonfatalCount,eventId,cause);
  record.woundEvents[record.woundEvents.length-1].injuryIds=injuries.map(i=>i.id);
  const penalty=eoeHealthWoundPenalty(id);
  // HP-01B3: machine-owned presentation facts. This does NOT create cards,
  // issue narration, alter combat, or broadcast undiscovered creature data.
  // Events are per-action; old entries never replay on subsequent turns.
  if (record.kind === "PC" || record.kind === "NPC") {
    const actionCount=typeof info!=="undefined" &&
      Number.isSafeInteger(info.actionCount) ? info.actionCount : null;
    if (!Array.isArray(eoeHealthStore().presentationEvents))
      eoeHealthStore().presentationEvents=[];
    eoeHealthStore().presentationEvents.push({id,kind:record.kind,
      name:record.kind==="PC" ? "Player" : record.name,
      applied,remaining,maxWounds:record.maxWounds,
      status:record.status,penalty,cause:cause||"unspecified",
      injuries:injuries.map(i=>({label:i.label,severity:i.severity})),
      actionCount,eventId:eventId||null});
    // Bounded metadata; historical events remain in the individual ledger.
    if (eoeHealthStore().presentationEvents.length>18)
      eoeHealthStore().presentationEvents.splice(0,eoeHealthStore().presentationEvents.length-18);
  }
  log(`HEALTH WOUND | id=${id} applied=${applied}/${wounds} remaining=${remaining}/${record.maxWounds} status=${record.status} penalty=${penalty} event=${eventId||"none"}`);
  return {ok:true,duplicate:false,applied,remaining,status:record.status,
    fatal,actionPenalty:penalty};
};

// ============================================================================
// HP-01C3 — DELIBERATE MEDICAL AID / SINGLE-WOUND TREATMENT
// This is an explicit, conservative bridging mechanic, not an attempt to
// reproduce ATDW's entire medical/time/consumables/infection system.
// Starting medkit: 3 one-use field dressings, granted ONCE per save, never
// replenished by Continue, visiting a Site, reloading, or expedition travel.
// No generic inventory or free-form natural-language healing is assumed.
// Any valid treatment attempt consumes ONE dressing (success or failure).
// Success restores one Wound and closes one active injury, if any. Existing
// unlabeled Wounds remain treatable. Dead PCs/NPCs cannot be resurrected.
// Critical injury specials remain deferred; no knockdown, time skip, or
// passive rest recovery exists. Do not promote injury labels to status flags.
// ============================================================================
const EOE_MEDICAL_SCHEMA_VERSION = 1;
const eoeMedicalStore = () => {
  if (state.eoeMedical === undefined) {
    const initial = ATDW_SETTINGS.INITIAL_FIELD_DRESSINGS;
    // Invalid user setting is a hard block, never a source of infinite stock.
    if (!Number.isSafeInteger(initial) || initial < 0 || initial > 20) {
      log("MEDICAL STOCK | invalid configured starting field dressings");
      return null;
    }
    state.eoeMedical = {schemaVersion:EOE_MEDICAL_SCHEMA_VERSION,
      charges:initial,attempts:{}};
    log(`MEDICAL STOCK | initial field dressings=${initial} (once per save)`);
  }
  const s=state.eoeMedical;
  if (!s || typeof s!=="object" || Array.isArray(s) ||
      s.schemaVersion!==EOE_MEDICAL_SCHEMA_VERSION ||
      !Number.isSafeInteger(s.charges) || s.charges < 0 || s.charges > 1000 ||
      !s.attempts || typeof s.attempts!=="object" || Array.isArray(s.attempts)) {
    log("MEDICAL STOCK | unsupported/corrupt medical schema; left unchanged");
    return null;
  }
  return s;
};
// One existing authoritative recovery path shared by field and station care.
// Called ONLY after a validated wound, living patient and event identity.
const eoeMedicalCloseOneWound = (patient,eventId) => {
  patient.woundsRemaining=Math.min(patient.maxWounds,patient.woundsRemaining+1);
  patient.status=patient.woundsRemaining===patient.maxWounds?"alive":"injured";
  const oldest=patient.injuries.find(i=>i && i.active!==false);
  if (!oldest) return null;
  oldest.active=false;
  oldest.recoveredBy=eventId;
  return typeof oldest.label==="string" ? oldest.label.slice(0,100) : null;
};
// MED-01B: station-only, per-docking-visit bridging service. Not a claim of
// canonical ATDW healing rates, fees or time costs. No automatic rest healing.
// Station supplies and treatment are explicit once per visit; a NEW visit
// requires a real return from another expedition (distinct Site identity).
const eoeAzaheimVisitKey = () => {
  if (!state.travel || state.travel.phase!=="idle" || state.currentSiteId) return null;
  const home=state.lastHomeArrival;
  const from=home && home.fromSiteId || state.lastCompletedSiteId;
  if (from!==undefined && from!==null) {
    const key=String(from);
    if (!/^[A-Za-z0-9_-]{1,90}$/.test(key)) return null;
    return `return:${key}`;
  }
  return "initial";
};
const eoeMedicalStationState = () => {
  const stock=eoeMedicalStore();
  const key=eoeAzaheimVisitKey();
  if (!stock || !key) return null;
  const previous=stock.station;
  if (previous!==undefined && (!previous || typeof previous!=="object" ||
      typeof previous.visitKey!=="string" ||
      typeof previous.resupplied!=="boolean" ||
      !previous.treated || typeof previous.treated!=="object" ||
      Array.isArray(previous.treated))) return null;
  if (!previous || previous.visitKey!==key) {
    stock.station={visitKey:key,resupplied:false,treated:{}};
  }
  return {stock,station:stock.station,key};
};
const eoeMedicalStationCommand = (operation,patientId,eventId) => {
  const error=message=>({ok:false,message:`AZAHEIM MEDICAL — NO ACTION\n${message}`});
  if (typeof eventId!=="string" || !eventId || eventId.length>160)
    return error("A valid action ID is required.");
  if (!eoeAzaheimVisitKey()) return error("Station care is available only at Azaheim, between expeditions.");
  const info=eoeMedicalStationState();
  if (!info) return error("Station medical state invalid; no supplies or Wounds changed.");
  const {stock,station}=info;
  if (operation==="RESUPPLY") {
    if (station.resupplied) return error("This docking visit's resupply was already collected.");
    const capacity=ATDW_SETTINGS.AZAHEIM_MEDKIT_CAPACITY;
    if (!Number.isSafeInteger(capacity) || capacity<1 || capacity>20)
      return error("Station resupply capacity is invalid.");
    if (stock.charges>=capacity) return error(`Already carrying ${stock.charges} dressings (station cap ${capacity}); resupply not used.`);
    const added=capacity-stock.charges;
    stock.charges=capacity;
    station.resupplied=true;
    return {ok:true,message:`AZAHEIM RESUPPLY — +${added} field dressings; ${stock.charges} remaining. Limited to once per docking visit. No healing occurred.`};
  }
  if (operation!=="CLINIC") return error("Unknown station medical service.");
  if (!["pc","crew:2","crew:3"].includes(patientId)) return error("Unknown patient.");
  if (station.treated[patientId]) return error("This patient has already received clinic care during this docking visit.");
  const patient=eoeHealthRecord(patientId);
  if (!eoeHealthValidWoundRecord(patient) || !["PC","NPC"].includes(patient.kind) ||
      !Array.isArray(patient.injuries) ||
      (patient.treatmentEvents!==undefined && !Array.isArray(patient.treatmentEvents)))
    return error("Invalid human patient or injury history; no healing occurred.");
  if (patient.status==="dead" || patient.woundsRemaining===0)
    return error("Death is permanent; clinic care cannot resurrect a patient.");
  if (patient.woundsRemaining>=patient.maxWounds)
    return error("Patient is healthy; the clinic visit was not used.");
  const closedInjury=eoeMedicalCloseOneWound(patient,eventId);
  station.treated[patientId]=true;
  if (!patient.treatmentEvents) patient.treatmentEvents=[];
  patient.treatmentEvents.push({eventId,healed:1,remaining:patient.woundsRemaining,
    success:true,closedInjury,chargesLeft:stock.charges,source:"azaheim_clinic"});
  return {ok:true,message:`AZAHEIM CLINIC — ${patientId==="pc"?"Player":patient.name} recovered 1 Wound; `+
    `${patient.woundsRemaining}/${patient.maxWounds} remaining; ${patient.status.toUpperCase()}. `+
    (closedInjury?`Injury treated: ${closedInjury}. `:"")+
    `No field dressings spent. Once per patient per docking visit.`};
};
const eoeMedicalDebugRefill = () => {
  const s=eoeMedicalStore();
  if (!s) return {ok:false,error:"Invalid medical stock; no change."};
  const old=s.charges;
  s.charges=3;
  return {ok:true,message:`DEBUG MEDKIT REFILL — ${old} -> ${s.charges} field dressings. Test fixture; no natural restock or healing.`};
};
const eoeHealthTreat = request => {
  const error=reason=>({ok:false,error:reason});
  if (!request || typeof request!=="object" || Array.isArray(request))
    return error("Invalid treatment request.");
  const {patientId,eventId}=request;
  if (typeof patientId!=="string" || !["pc","crew:2","crew:3"].includes(patientId))
    return error("Unknown patient. Use Player, Bianca or Sam; unrated creatures cannot be treated.");
  if (typeof eventId!=="string" || !eventId.trim() || eventId.length>160)
    return error("A unique treatment action identity is required.");
  const stock=eoeMedicalStore();
  if (!stock) return error("Medical inventory state is unavailable or unsupported.");
  if (Object.prototype.hasOwnProperty.call(stock.attempts,eventId))
    return {...stock.attempts[eventId],duplicate:true};
  const patient=eoeHealthRecord(patientId);
  if (!eoeHealthValidWoundRecord(patient) || !["PC","NPC"].includes(patient.kind))
    return error("Patient has no valid human Wound record.");
  if (patient.status==="dead" || patient.woundsRemaining===0)
    return error("Patient is dead; treatment cannot resurrect them.");
  if (patient.woundsRemaining===patient.maxWounds)
    return error("Patient has no Wounds to heal; no supplies consumed.");
  const canAct=eoeHealthCanAct("pc");
  if (!canAct.canAct) return error("The player cannot perform treatment.");
  const healer=state.eoePlayerProfile;
  if (!healer || healer.kind!=="PC")
    return error("A completed player skill profile is required for Medical Aid.");
  const woundPenalty=eoeHealthWoundPenalty("pc");
  if (woundPenalty===null) return error("Invalid healer health; no check rolled.");
  // Validate the intended check BEFORE consuming a dressing or rolling.
  if (!healer.skills || !Number.isSafeInteger(healer.skills["Medical Aid"]))
    return error("Medical Aid skill rating is unavailable; no supplies consumed.");
  if (!Array.isArray(patient.injuries) ||
      (patient.treatmentEvents!==undefined && !Array.isArray(patient.treatmentEvents)))
    return error("Invalid treatment history; no supplies consumed.");
  if (stock.charges<1)
    return error("No field dressings remain. Treatment cannot be attempted.");
  // A deterministic natural-20 injection is reserved for the explicit debug
  // command. Ordinary #TREAT always rolls normally. No public dice argument.
  if (request.debugRolls!==undefined &&
      (!request.debugAuthorized || !Array.isArray(request.debugRolls) ||
       request.debugRolls.length!==1 || request.debugRolls[0]!==20))
    return error("Invalid debug treatment request; no supplies consumed.");
  const check=eoeResolveCheck({actor:healer,check:"Medical Aid",difficulty:"normal",
    woundPenalty,...(request.debugAuthorized ? {rolls:request.debugRolls} : {})});
  if (!check.ok) return error(`Medical Aid check unavailable: ${check.error}`);
  const before=patient.woundsRemaining;
  stock.charges--;
  let closedInjury=null;
  if (check.success) {
    closedInjury=eoeMedicalCloseOneWound(patient,eventId);
  }
  const result={ok:true,duplicate:false,eventId,patientId,
    patientName:patientId==="pc"?"Player":patient.name,
    check,healed:check.success?1:0,remaining:patient.woundsRemaining,
    maxWounds:patient.maxWounds,status:patient.status,
    closedInjury,chargesLeft:stock.charges};
  // Retain action ledger to protect against Retry/replayed event identities.
  stock.attempts[eventId]=result;
  if (!patient.treatmentEvents) patient.treatmentEvents=[];
  patient.treatmentEvents.push({eventId,healed:result.healed,remaining:result.remaining,
    success:check.success,closedInjury,chargesLeft:stock.charges});
  log(`MEDICAL TREAT | patient=${patientId} success=${check.success} healed=${result.healed} ${before}->${result.remaining}/${patient.maxWounds} dressings=${stock.charges} event=${eventId}`);
  return result;
};
const eoeHealthTreatmentReport = turn => {
  if (!turn) return "";
  if (!turn.ok) return `TREATMENT REJECTED — ${turn.error}`;
  const r=turn.check;
  const outcome=r.success?"SUCCESS":"FAILURE";
  return `TREATMENT RESULT — ${turn.patientName}\n`+
    `Medical Aid: D20 ${r.die} + ${r.score}`+
    (r.woundPenalty<0?` - ${Math.abs(r.woundPenalty)} (Wounds)`:"")+
    ` = ${r.total} vs ${r.target} — ${outcome}`+
    (r.critical?` (critical ${r.critical})`:"")+"\n"+
    `Recovery: ${turn.healed} Wound${turn.healed===1?"":"s"} healed; `+
    `${turn.remaining}/${turn.maxWounds} Wounds remaining; status ${turn.status.toUpperCase()}.`+
    (turn.closedInjury?` Injury treated: ${turn.closedInjury}.`:"")+"\n"+
    `Field dressings left: ${turn.chargesLeft}.`;
};

// ============================================================================
// HP-01B2 — TERMINAL DEATH, NO INCAPACITATION/TURN SKIPPING
//
// Consciousness/knockdown mechanics are deliberately deferred. A living
// wounded character always retains control, and a dead PC ends the scenario's
// playable lifecycle. AI Dungeon has no documented "end adventure" function:
// Input blocks mechanics, Context refuses revival, and Output deterministically
// replaces further AI continuations with GAME OVER. Avoid stop:true; it raises
// scripting errors in Input according to the current API documentation.
// ============================================================================
const eoeHealthCanAct = id => {
  if (typeof id !== "string" || !id.trim())
    return {ok:false,canAct:false,reason:"invalid actor id"};
  const r=eoeHealthRecord(id);
  if (!r) return {ok:false,canAct:false,reason:"unknown health record"};
  // A zero-Wound record is terminal even if a damaged save has an inconsistent
  // status string. This fails safely, without mutating older health records.
  if (r && r.schemaVersion === EOE_HEALTH_SCHEMA_VERSION &&
      r.kind === "PC" && r.woundsRemaining === 0)
    return {ok:true,canAct:false,reason:"dead"};
  if (!eoeHealthValidWoundRecord(r))
    return {ok:false,canAct:false,reason:"invalid health record"};
  if (r.status === "dead" || r.woundsRemaining === 0)
    return {ok:true,canAct:false,reason:"dead"};
  // No persistent "incapacitated" state in HP-01B2. An old experimental
  // boolean is ignored rather than causing an infinite Continue-only loop.
  return {ok:true,canAct:true,reason:"ready"};
};
const eoeHealthGameOverText = () => {
  const pc=eoeHealthRecord("pc");
  const events=pc && Array.isArray(pc.woundEvents) ? pc.woundEvents : [];
  const last=events.length ? events[events.length-1] : null;
  const cause=last && last.cause && last.cause!=="unspecified"
    ? `\nFinal injury: ${eoeHealthSafeNarrativeCause(last.cause)}.` : "";
  return "GAME OVER — YOUR PATHFINDER HAS DIED\n"+
    "The expedition ends here. No further actions can be taken in this adventure."+cause;
};

// ============================================================================
// SKL-01A — ATDW-BASED CHECK ENGINE (mechanics only, no automatic rolls)
// The Eirseid setting renames the source skill "Àrsaidh Technology" as
// "Eirseid Technology". All other core modifiers follow the tabletop rules.
// These pure helpers intentionally do not create character sheets in state.
// Caller must register real scores; never infer numerical stats from prose.
// ============================================================================
const EOE_SKILL_NAMES = [
  "Eirseid Technology", "Close Combat", "Manipulation", "Medical Aid",
  "Perception", "Pilot", "Ranged Combat", "Resolve", "Science",
  "Stealth", "Survival", "Technology"
];
const EOE_ATTRIBUTE_NAMES = ["STR", "DEX", "CON", "WIL", "INT", "CHA"];
const EOE_DIFFICULTY_MODIFIERS = {
  "childs play": 15, "effortless": 10, "easy": 5, "normal": 0,
  "demanding": -5, "hard": -10, "almost impossible": -15
};
const EOE_NPC_SKILL_ATTRIBUTE = {
  "Eirseid Technology": "INT", "Close Combat": "STR", "Manipulation": "CHA",
  "Medical Aid": "DEX", "Perception": "WIL", "Pilot": "DEX",
  "Ranged Combat": "DEX", "Resolve": "WIL", "Science": "INT",
  "Stealth": "DEX", "Survival": "WIL", "Technology": "INT"
};
const eoeSkillError = reason => ({ ok: false, error: reason });
const eoeIsFiniteInteger = n => typeof n === "number" && Number.isInteger(n) && Number.isFinite(n);
const eoeCanonicalCheck = raw => {
  const input = String(raw == null ? "" : raw).trim();
  if (EOE_ATTRIBUTE_NAMES.includes(input.toUpperCase())) return input.toUpperCase();
  const normalized = input.toLowerCase().replace(/\s+/g, " ");
  if (normalized === "àrsaidh technology" || normalized === "arsaidh technology")
    return "Eirseid Technology";
  return EOE_SKILL_NAMES.find(name => name.toLowerCase() === normalized) || null;
};
/**
 * Create initial PC scores, using ATDW's 70 skill allocation points, max
 * 10 allocated to a single skill. Initial Eirseid Technology base is -5.
 * Primary attributes: six scores start at 8 plus 12 points, max 18.
 * This is an *optional validator*, not an automatic character generator.
 */
const eoeBuildStartingSkillSheet = (allocations, attributes) => {
  if (!allocations || typeof allocations !== "object" || Array.isArray(allocations) ||
      !attributes || typeof attributes !== "object" || Array.isArray(attributes))
    return eoeSkillError("Expected skill allocations and attribute maps.");
  if (Object.keys(allocations).length !== EOE_SKILL_NAMES.length ||
      !Object.keys(allocations).every(k => EOE_SKILL_NAMES.includes(k)))
    return eoeSkillError("Provide exactly the 12 ATDW skill allocations.");
  let skillPoints = 0;
  const scores = {};
  for (const name of EOE_SKILL_NAMES) {
    const n = allocations[name];
    if (!eoeIsFiniteInteger(n) || n < 0 || n > 10)
      return eoeSkillError(`Invalid starting allocation for ${name}: expected 0..10.`);
    skillPoints += n;
    scores[name] = n + (name === "Eirseid Technology" ? -5 : 0);
  }
  if (skillPoints !== 70)
    return eoeSkillError(`Starting skill allocation must total 70, found ${skillPoints}.`);
  if (Object.keys(attributes).length !== EOE_ATTRIBUTE_NAMES.length ||
      !Object.keys(attributes).every(k => EOE_ATTRIBUTE_NAMES.includes(k)))
    return eoeSkillError("Provide exactly six ATDW primary attributes.");
  let attributeTotal = 0;
  const attrScores = {};
  for (const name of EOE_ATTRIBUTE_NAMES) {
    const n = attributes[name];
    if (!eoeIsFiniteInteger(n) || n < 8 || n > 18)
      return eoeSkillError(`Invalid initial ${name}: expected 8..18.`);
    attributeTotal += n;
    attrScores[name] = n;
  }
  if (attributeTotal !== 60)
    return eoeSkillError(`Starting attributes must total 60 (48 + 12), found ${attributeTotal}.`);
  return { ok: true, kind: "PC", skills: scores, attributes: attrScores };
};
/**
 * Resolve a check with no state mutation. "rolls" is for deterministic
 * testing; production callers omit it and use Math.random. A failed lookup
 * returns ok:false, not an invented score of zero. The rolled die—not the
 * total—determines natural-1 and natural-20 criticals.
 */
const eoeResolveCheck = (options = {}) => {
  const actor = options.actor;
  const check = eoeCanonicalCheck(options.check);
  if (!check) return eoeSkillError("Unknown ATDW skill or attribute.");
  if (!actor || !["PC", "NPC"].includes(actor.kind))
    return eoeSkillError("Expected a registered PC or NPC actor.");
  const checkType = EOE_ATTRIBUTE_NAMES.includes(check) ? "attribute" : "skill";
  let scoreKey = check;
  let scores = checkType === "attribute" ? actor.attributes : actor.skills;
  if (actor.kind === "NPC" && checkType === "skill") {
    scoreKey = EOE_NPC_SKILL_ATTRIBUTE[check];
    scores = actor.attributes;
  }
  const score = scores && Object.prototype.hasOwnProperty.call(scores, scoreKey)
    ? scores[scoreKey] : undefined;
  if (!eoeIsFiniteInteger(score))
    return eoeSkillError(`No numeric ${scoreKey} score registered for this actor.`);
  const difficultyName = String(options.difficulty == null ? "normal" : options.difficulty)
    .toLowerCase().replace(/[’']/g, "").trim().replace(/\s+/g, " ");
  if (!Object.prototype.hasOwnProperty.call(EOE_DIFFICULTY_MODIFIERS, difficultyName))
    return eoeSkillError("Unknown ATDW difficulty.");
  const difficultyModifier = EOE_DIFFICULTY_MODIFIERS[difficultyName];
  const situational = options.situational === undefined ? 0 : options.situational;
  if (!eoeIsFiniteInteger(situational) || Math.abs(situational) > 100)
    return eoeSkillError("Situational modifier must be a finite integer in -100..100.");
  // HP-01C1: separate authoritative Wound penalty from difficulty, skill,
  // assistance and situational modifiers. The resolver is PURE: callers must
  // read the registered health record and pass this penalty exactly once.
  // Never add it from another path or infer it from a Story Card.
  const woundPenalty = options.woundPenalty === undefined ? 0 : options.woundPenalty;
  if (!eoeIsFiniteInteger(woundPenalty) || woundPenalty > 0 || woundPenalty < -100)
    return eoeSkillError("Wound penalty must be an integer in -100..0.");
  const mode = options.mode == null ? "normal" : options.mode;
  if (!["normal", "advantage", "disadvantage"].includes(mode))
    return eoeSkillError("Mode must be normal, advantage, or disadvantage.");
  const needed = mode === "normal" ? 1 : 2;
  let rolls;
  if (options.rolls !== undefined) {
    if (!Array.isArray(options.rolls) || options.rolls.length !== needed ||
        !options.rolls.every(r => eoeIsFiniteInteger(r) && r >= 1 && r <= 20))
      return eoeSkillError("Injected die values must be exactly 1..20, one or two rolls as required.");
    rolls = options.rolls.slice();
  } else {
    rolls = Array.from({length: needed}, () => Math.floor(Math.random() * 20) + 1);
  }
  const die = mode === "advantage" ? Math.max(...rolls)
    : mode === "disadvantage" ? Math.min(...rolls) : rolls[0];
  const assistModifier = options.assist === true && checkType === "skill" &&
    check !== "Close Combat" && check !== "Ranged Combat" ? 2 : 0;
  const total = die + score + difficultyModifier + situational + assistModifier + woundPenalty;
  const critical = die === 1 ? "failure" : die === 20 ? "success" : null;
  const success = critical === "success" || (critical !== "failure" && total >= 20);
  const level = success && total >= 20
    ? (total >= 36 ? "Incredible" : total >= 31 ? "Impressive" : total >= 25 ? "Great" : "Normal")
    : null;
  return {
    ok: true, actorKind: actor.kind, check, checkType, sourceAttribute: scoreKey,
    score, difficulty: difficultyName, difficultyModifier, situational,
    assistModifier, woundPenalty, mode, rolls, die, total, target: 20, critical, success, level
  };
};

// ============================================================================
// SKL-01B3 — EXPLICIT SKILL CHECKS (ATDW D20; NO IMPLICIT DETECTION)
// Player supplies only an actor and a skill/attribute. Difficulty and other
// modifiers are NOT player-authorized: until challenge-specific difficulties
// are implemented, checks are Normal and unassisted. No extra actions, resource
// spending, injury, artifact activation, or scene-state mutation is inferred.
// A single input action creates at most one durable d20 result. Context then
// supplies a compact immutable outcome seed; Output displays the actual roll.
// ============================================================================
const eoeSkillCheckCommandPresent = raw =>
  /(?:^|[^A-Za-z0-9_#])#skillcheck\b/i.test(String(raw || ""));
const eoeParseSkillCheckRequest = raw => {
  const input=String(raw || "");
  const tokens=Array.from(input.matchAll(/#([A-Za-z][A-Za-z0-9_]*)/g));
  // Reject combinations such as #skillcheck Science #LIFTOFF before either
  // system mutates state; commands must be a single unambiguous action.
  if (tokens.length !== 1 || tokens[0][1].toLowerCase() !== "skillcheck")
    return eoeSkillError("Multiple or conflicting #commands. Use exactly one #skillcheck command per action.");
  const marker=tokens[0];
  const at=marker.index;
  if (at > 0 && /[A-Za-z0-9_#]/.test(input[at-1]))
    return eoeSkillError("Invalid #skillcheck command boundary.");
  let tail=input.slice(at+marker[0].length).trim().replace(/[.!?]+\s*$/,"").trim();
  if (!tail) return eoeSkillError("Usage: #skillcheck Technology or #skillcheck Bianca Science. Choose a named ATDW skill or attribute.");
  let check=null, actorToken=null;
  // Match the full, exact trailing skill first (longer names before shorter).
  const all=EOE_SKILL_NAMES.concat(EOE_ATTRIBUTE_NAMES).sort((a,b)=>b.length-a.length);
  for (const candidate of all) {
    const escaped=candidate; // ATDW check names contain no regex metacharacters.
    const re=new RegExp(`(?:^|\\s)(${escaped})$`,"i");
    const match=tail.match(re);
    if (!match) continue;
    check=eoeCanonicalCheck(candidate);
    actorToken=tail.slice(0,match.index).trim();
    break;
  }
  if (!check) return eoeSkillError("Unknown or malformed skill check. Use a complete ATDW skill/attribute name; difficulty overrides are not accepted.");
  // Text before the command is an attempted action, not a license to choose
  // unmentioned player behavior. Omit leading Say/Do UI prefixes.
  const attempt=input.slice(0,at).replace(/^\s*>\s*(?:You\s+)?/i,"").trim().slice(0,240);
  return {ok:true,check,actorToken,attempt};
};
// SKL-01C — Deterministic, player-agency-safe procedural scene plan.
// These are binding narrative facts and limits, not canned player dialogue.
// Context hands them to the AI for prose expansion after JavaScript resolves
// the roll. Never infer a new voluntary player action from check success/failure.
const eoeBuildSkillNarrativePlan = (result, attempt, actorKind) => {
  const specified=!!String(attempt || "").trim();
  const outcome=specified
    ? (result.success ? "The stated objective is achieved." : "The stated objective is not achieved; the obstacle remains unresolved.")
    : (result.success ? "The check succeeds mechanically, but no specific attempted activity was described."
      : "The check fails mechanically, but no specific attempted activity was described.");
  const immediate=specified
    ? "Narrate only the immediate execution of the specified attempt and its direct perceptible effects; do not invent additional damage, resources, movements, or new objectives."
    : "No specific action is authorized: describe only the roll and immediate scene impressions, not an invented repair, attack, theft, movement, or task.";
  const boundary=actorKind==="NPC"
    ? "NPCs retain independent agency. Do not invent the player's speech, consent, follow-up actions, or next decision. Leave player participation unspecified."
    : "No new decision, player speech, voluntary reaction, or follow-up action is authorized. A failure does not mean the player gives up. Return control to the player.";
  const critical=result.critical==="failure"
    ? "Critical failure does not itself authorize an unmodeled wound, item destruction, inventory loss, or permanent state change."
    : result.critical==="success"
    ? "Critical success does not authorize extra inventory, hidden knowledge, or a new unscripted activity."
    : "No critical consequence is specified.";
  return {outcome,immediate,boundary,critical,attemptSpecified:specified};
};
// HP-01C1: look up a PC or NPC's authoritative Wound penalty. Normal Input
// always registers party health before checks. Only legacy test/internal
// callers with NO health registry can use the pre-health zero fallback.
// If a registry exists but is damaged, never silently award a healthy roll.
const eoeSkillActorWoundPenalty = actor => {
  const id=actor && actor.kind==="NPC" ? `crew:${actor.slot}` : "pc";
  if (state.eoeHealth === undefined) return {ok:true,value:0};
  const penalty=eoeHealthWoundPenalty(id);
  if (penalty === null) return {ok:false,error:`Invalid or missing authoritative health record for ${id}; check not rolled.`};
  return {ok:true,value:penalty};
};
const eoeResolveExplicitSkillCheck = (raw, actionId) => {
  const parsed=eoeParseSkillCheckRequest(raw);
  if (!parsed.ok) return {ok:false,actionId,error:parsed.error};
  let actor=null, actorName="Player";
  const token=(parsed.actorToken || "").trim().toLowerCase();
  if (!token || ["me","myself","self","player"].includes(token)) {
    actor=state.eoePlayerProfile;
    if (!actor || actor.kind !== "PC")
      return {ok:false,actionId,error:"Player skill profile unavailable. Complete #SKILLSETUP before rolling."};
  } else {
    const choices=[2,3].map(slot=>state.eoeCrewProfiles && state.eoeCrewProfiles[slot]).filter(Boolean);
    const matches=choices.filter(p=>p.kind==="NPC" &&
      (String(p.name||"").toLowerCase()===token || `crew${p.slot}`===token));
    if (matches.length !== 1)
      return {ok:false,actionId,error:"Unknown or unavailable crew actor. Use the full name of a registered companion."};
    actor=matches[0];actorName=String(actor.name);
  }
  // Terminal NPC death forbids acting, even if the storyteller attempts it.
  const actorHealth=eoeHealthRecord(actor.kind==="NPC" ? `crew:${actor.slot}` : "pc");
  if (actorHealth) {
    const eligibility=eoeHealthCanAct(actor.kind==="NPC" ? `crew:${actor.slot}` : "pc");
    if (!eligibility.canAct)
      return {ok:false,actionId,error:`${actorName} is ${eligibility.reason}; skill check not rolled.`};
  }
  const wound=eoeSkillActorWoundPenalty(actor);
  if (!wound.ok) return {ok:false,actionId,error:wound.error};
  const result=eoeResolveCheck({actor,check:parsed.check,difficulty:"normal",
    woundPenalty:wound.value});
  if (!result.ok) return {ok:false,actionId,error:result.error};
  const narrativePlan=eoeBuildSkillNarrativePlan(result,parsed.attempt,actor.kind);
  // Retain the legacy `seed` property for compatibility with existing saves
  // and callers that consume the fixed summary as a string.
  const seed=[narrativePlan.outcome,narrativePlan.critical].join(" ");
  return {ok:true,actionId,actorName,attempt:parsed.attempt,check:result.check,result,
    seed,narrativePlan,source:"EXPLICIT",schemaVersion:1};
};
const eoeSkillCheckResultText = turn => {
  if (!turn || !turn.ok) return turn && turn.error ? `SKILL CHECK NOT ROLLED — ${turn.error}` : "";
  const r=turn.result;
  const modifier=r.score+r.difficultyModifier+r.situational+r.assistModifier;
  const roll=r.rolls.length>1 ? r.rolls.join(" / ") : String(r.die);
  const sign=modifier<0?"-":"+";
  // Keep Wounds legible rather than silently folding them into the skill score.
  const woundLabel=r.woundPenalty<0 ? ` - ${Math.abs(r.woundPenalty)} (Wounds)` : "";
  const label=r.success?`SUCCESS${r.level?` (${r.level})`:""}`:"FAILURE";
  const crit=r.critical?` — CRITICAL ${r.critical.toUpperCase()}`:"";
  const linked=r.actorKind==="NPC" && r.checkType==="skill" ? ` [${r.sourceAttribute}]`:"";
  return `SKILL CHECK RESULT — ${turn.actorName}: ${r.check}${linked}\n`+
    `D20 ${roll} (kept ${r.die}) ${sign} ${Math.abs(modifier)}${woundLabel} = ${r.total} vs ${r.target} — ${label}${crit}`;
};

// ============================================================================
// SKL-01D — CONSERVATIVE IMPLICIT SKILL ACTION DETECTOR
// Pure single-turn recognizer. Favors precision over recall: only immediately
// attempted, player-authored tasks with a credible mechanical challenge.
// Abstain on normal look/talk, speculation, quoted dialogue, NPC instructions,
// mixed/unclear tasks, and combat (reserved for CBT-01). Commands take priority.
// No AI-generated classifications, story history, or NPC narration are trusted.
// ============================================================================
const eoeDetectImplicitSkillCheckRequest = raw => {
  const original=String(raw == null ? "" : raw).trim();
  if (!original || original.length>600 || /#[A-Za-z]/.test(original)) return null;
  const uncluttered=original.replace(/^>\s*(?:You\s+)?/i,"").trim();
  const t=uncluttered.toLowerCase().replace(/\s+/g," ");
  if (!t || /^['"“‘]/.test(t) || /\?/.test(t) || /^(?:i\s+)?set up an appointment\b/i.test(t)) return null;
  // A player giving a companion an instruction is not that companion taking
  // the action. Pure speech, quotations, instructions and hypothetical plans
  // must not roll. An explicit #skillcheck remains available for edge cases.
  if (/\b(?:don't|do not|never|won't|wouldn't|cannot|can't|not going to|refuse to|instead of|rather than)\b/i.test(t) ||
      /\b(?:maybe|perhaps|might|could|should|would|consider|thinking of|plan to|want to|wish|someday|later|tomorrow|if i|what if)\b/i.test(t) ||
      /^(?:i|we)\s+(?:ask|tell|order|instruct|request|command|suggest|quote|describe|report|remember|think|wonder|discuss|read|write|draw|observe|greet|listen|glance|stand|sit|watch|check on|point)\b/i.test(t) ||
      /^(?:bianca|sam|the\s+\w+|someone|anyone|he|she|they|you)\b/i.test(t) ||
      /^\w+(?:\s+\w+){0,2},\s*(?:please\s+)?(?:repair|hack|analyze|fix|treat|sneak|help|pilot|persuade|decode)\b/i.test(t)) return null;
  // Require the player's own action, or an unequivocal Do-mode imperative.
  // Avoid rolling for unrelated exposition with a skill word buried inside.
  const actor=/^(?:i\s+|we\s+|my\s+)?(?:try\s+to\s+|attempt\s+to\s+|attempt\s+|carefully\s+|quietly\s+|quickly\s+|use\s+(?:my|a|an|the)\s+(?:[\w-]+\s+){1,4}to\s+)?/i;
  const stripped=t.replace(actor,"");
  const firstWord=/^(?:repair|restore|jury-rig|reconnect|reboot|operate|unlock|translate|interpret|diagnose|fix|rewire|reroute|recalibrate|calibrate|troubleshoot|splice|solder|hack|override|patch|interface|decode|decipher|reprogram|activate|bypass|analy[sz]e|conduct|determine|synthesize|identify|assay|test|classify|study|perform|stitch|seal|administer|stabilize|diagnose|check|set|treat|bandage|resuscitate|sneak|conceal|creep|hide|slip|evade|move|pickpocket|persuade|threaten|convince|bluff|intimidate|negotiate|bargain|lie|coax|search|inspect|look|spot|detect|examine|scan|locate|track|forage|follow|find|build|navigate|pilot|guide|steer|dock|maneuver|resist|keep|push|pry|force|lift|bend|pick|disarm)\b/i;
  if (!firstWord.test(stripped)) return null;
  // No navigating site passages, implied combat, or high-confidence movement
  // command. Do not intercept the graph-navigation subsystem.
  if (/\b(?:attack|shoot|stab|strike|punch|fire at|kill|swing|fight)\b/i.test(t) ||
      /\b(?:exit\s*\d+|area\s*\d+|enter the room|enter the corridor)\b/i.test(t) ||
      /\b(?:walk|head|go|enter|descend|ascend|cross|return|backtrack)\s+(?:to|through|down|up|into|along|toward|for)\b/i.test(stripped)) return null;

  // Rules are purposefully small, testable groups of verbs and objects.
  // Multiple independently matching categories indicate ambiguity, not a
  // license to choose whichever mechanical roll benefits the player.
  const categories=[];
  const has=(re)=>re.test(t);
  const add=(name,verb,context)=>{if(has(verb)&&(!context||has(context)))categories.push(name);};
  const technicalVerb=/\b(?:repair|restore|jury-rig|reconnect|reboot|operate|unlock|translate|interpret|diagnose|fix|rewire|reroute|recalibrate|calibrate|troubleshoot|splice|solder|hack|override|patch|interface|decode|decipher|reprogram|activate|bypass)\b/i;
  const alien=/\b(?:eirseid|alien|àrsaidh|arsaidh)\b/i;
  const technicalObject=/\b(?:console|circuit|panel|purifier|cooling|recycler|terminal|generator|wiring|wire|sensor|antenna|electronics|machinery|machine|junction box|wires|device|security|communications|array|system|computer|mechanism|motor|power|processor|board|bridge|display|screen|monitor|readout)\b/i;
  if (has(technicalVerb)&&has(alien))categories.push('Eirseid Technology');
  else add('Technology',technicalVerb,technicalObject);
  add('Science',/\b(?:analy[sz]e|analysis|identify|conduct|determine|synthesize|assay|test|classify|study)\b/i,
    /\b(?:chemical|compound|biological|specimen|sample|fluid|species|fungal|tissue|toxin|toxicity|toxins|organism|dna|radiation|spore|spores|blood|contamination|molecular|microscope|reagent|substance|formula|assay)\b/i);
  add('Medical Aid',/\b(?:stitch|seal|administer|suture|stabilize|diagnose|check|set|treat|bandage|resuscitate|cpr)\b/i,
    /\b(?:wound|first aid|pulse|bleeding|injured|injury|crewmate|infection|arm|leg|fracture|patient|cpr|unconscious|shoulder|man|woman|him|her)\b/i);
  add('Stealth',/\b(?:sneak|conceal|creep|hide|slip|evade|pickpocket|move silently)\b/i,
    /\b(?:past|from|behind|guard|sentry|creature|patrol|unseen|silently|spotted|pocket|enemy|shadow|shadows|sight)\b/i);
  add('Manipulation',/\b(?:persuade|threaten|convince|bluff|intimidate|negotiate|bargain|lie to|coax)\b/i,
    /\b(?:clerk|guard|smuggler|trader|merchant|officer|customs|stranger|captain|crew|him|her|them|npc|price|fee|trade|deal|past|secret)\b/i);
  add('Perception',/\b(?:search|inspect|look|spot|detect|examine|scan|locate)\b/i,
    /\b(?:hidden|concealed|trap|traps|tampering|pressure plate|ambush|compartment|wires|tripwire|tripwires|secret)\b/i);
  add('Survival',/\b(?:track|forage|follow|find|locate|build|navigate)\b/i,
    /\b(?:creature|predator|spoor|canyon|animal tracks|wilderness|edible|berries|forest|plants|mud|potable water|desert|drinking water|wastes|wasteland|shelter|acid rain|trail|food)\b/i);
  add('Pilot',/\b(?:pilot|guide|steer|dock|maneuver|land)\b/i,
    /\b(?:shuttle|craft|debris|transport ship|asteroid field|bay|vessel|ship)\b/i);
  add('Resolve',/\b(?:resist|keep|push through|endure)\b/i,
    /\b(?:psychic|intrusion|mind|composure|terror|panic|mental|fear)\b/i);
  add('STR',/\b(?:pry|force|lift|bend|break)\b/i,
    /\b(?:jammed|sealed|heavy|stuck|hatch|door|crate|grate|bulkhead|gate)\b/i);
  add('DEX',/\b(?:pick|disarm)\b/i,/\b(?:mechanical lock|spring-loaded trap|trap|lock)\b/i);
  // If multiple skills match, leave it to explicit #skillcheck or ordinary
  // narration, never perform two different checks from a single action.
  const options=[...new Set(categories)];
  if(options.length!==1) return null;
  return {ok:true,check:options[0],attempt:uncluttered.slice(0,240),source:'IMPLICIT',confidence:'high'};
};
const eoeResolveImplicitSkillCheck = (raw, actionId, parsed) => {
  if (!parsed || !parsed.ok || !state.eoePlayerProfile || state.eoePlayerProfile.kind!=='PC') return null;
  const wound=eoeSkillActorWoundPenalty(state.eoePlayerProfile);
  if (!wound.ok) return null; // conservative detector: no invented numeric result
  const result=eoeResolveCheck({actor:state.eoePlayerProfile,check:parsed.check,
    difficulty:'normal',woundPenalty:wound.value});
  if(!result.ok)return null;
  const narrativePlan=eoeBuildSkillNarrativePlan(result,parsed.attempt,'PC');
  return {ok:true,actionId,actorName:'Player',attempt:parsed.attempt,check:result.check,
    result,seed:[narrativePlan.outcome,narrativePlan.critical].join(' '),narrativePlan,
    source:'IMPLICIT',schemaVersion:1};
};

// ============================================================================
// SKL-01B1 — LIGHTWEIGHT NUMBERED CHARACTER SETUP (NO ROLLS YET)
// Scenario editor remains Opening: STORY with its original placeholders.
// On a NEW adventure, the first Input turn begins skill selection. Existing
// save files are not interrupted; #SKILLSETUP is optional at Azaheim.
// Selection commands: 1 / option 1 / #CHOOSE1 / #CHOOSE 1, one integer only.
// There is one fixed 1..12 list at EVERY step; duplicates are rejected.
// The setup chooses 10, 8, 7 allocated; nine unselected skills each get 5.
// This preserves ATDW's 70 starting points and the 10-point allocation cap.
// Balanced primary attributes are 10 each; attribute customization is later.
// ============================================================================
const EOE_SETUP_RANKS = ["PRIMARY", "SECONDARY", "TERTIARY"];
const EOE_SETUP_ALLOCATIONS = [10, 8, 7];
const EOE_PC_PROFILE_TYPE = "EOE_PLAYER_PROFILE";
const EOE_PC_PROFILE_KEY = "eoe_pc_skill_profile";
const EOE_PC_PROFILE_MARKER = "EOE_PROFILE_ID: PC";
const eoeNewSkillSetup = () => ({phase: "primary", picks: [], lastActionId: null, lastResponse: null});
const eoeSetupProfileText = (profile) => {
  const attribs = EOE_ATTRIBUTE_NAMES.map(k => `${k} ${profile.attributes[k]}`).join(" | ");
  const skills = EOE_SKILL_NAMES.map(k => `${k}: ${profile.skills[k]}`).join("\n");

};
const eoeEnsurePlayerProfileCard = () => {
  const profile = state.eoePlayerProfile;
  if (!profile || profile.kind !== "PC") return false;
  const cards = typeof storyCards !== "undefined" && Array.isArray(storyCards) ? storyCards : [];
  const idx = cards.findIndex(c => c.type === EOE_PC_PROFILE_TYPE &&
    String(c.entry || "").includes(EOE_PC_PROFILE_MARKER));
  const entry = eoeSetupProfileText(profile);
  if (idx >= 0) {
    if (cards[idx].entry !== entry || cards[idx].keys !== EOE_PC_PROFILE_KEY)
      updateStoryCard(idx, EOE_PC_PROFILE_KEY, entry, EOE_PC_PROFILE_TYPE, "PLAYER — ATTRIBUTES & SKILLS");
    return true;
  }
  if (cards.some(c => c.keys === EOE_PC_PROFILE_KEY)) {
    log("SKL CARD | key collision, unrelated card preserved");
    return false;
  }
  const added = addStoryCard(EOE_PC_PROFILE_KEY, entry, EOE_PC_PROFILE_TYPE, "PLAYER — ATTRIBUTES & SKILLS");
  if (added === false) { log("SKL CARD | creation failed"); return false; }
  log("SKL CARD | created PC profile");
  return true;
};
/**
 * Input validation is script-controlled; never ask the AI to guess a number.
 * A single exactly-numbered choice is accepted, never an action plus a choice.
 * Never assign a score until all three distinct picks have been validated.
 */
const eoeTakeSkillSetupChoice = (raw, actionId) => {
  const setup = state.eoeSkillSetup;
  if (!setup || setup.phase === "complete") return {ok:false, reason:"Setup is not active."};
  // Re-executing Input on Retry must not consume another ranked choice.
  if (actionId != null && setup.lastActionId === actionId) return setup.lastResponse;
  const input = String(raw || "").trim()
    .replace(/^>\s*(?:You\s+)?/i, "").trim();
  const match = input.match(/^(?:#(?:CHOOSE|SELECT)\s*|(?:OPTION|CHOICE)\s+)?([0-9]+)[.!]?$/i);
  let response;
  if (!match) {
    response = {ok:false, reason:"Enter one number from 1 to 12 (for example, #CHOOSE 4). Other actions are paused during setup."};
  } else {
    const number = Number(match[1]);
    if (!Number.isSafeInteger(number) || number < 1 || number > EOE_SKILL_NAMES.length) {
      response = {ok:false, reason:`Invalid selection ${match[1]}. Enter a number from 1 to ${EOE_SKILL_NAMES.length}.`};
    } else {
      const name = EOE_SKILL_NAMES[number - 1];
      if (setup.picks.includes(name)) {
        response = {ok:false, reason:`${name} is already selected. Choose a DIFFERENT skill.`};
      } else {
        setup.picks.push(name);
        const rank = EOE_SETUP_RANKS[setup.picks.length - 1];
        if (setup.picks.length === 3) {
          const allocations = {};
          for (const skill of EOE_SKILL_NAMES) allocations[skill] = 5;
          setup.picks.forEach((skill,i) => { allocations[skill] = EOE_SETUP_ALLOCATIONS[i]; });
          const attrs = {};
          for (const attr of EOE_ATTRIBUTE_NAMES) attrs[attr] = 10;
          const sheet = eoeBuildStartingSkillSheet(allocations, attrs);
          if (!sheet.ok) {
            setup.picks.pop();
            response = {ok:false, reason:`Unable to construct a valid character sheet: ${sheet.error}`};
          } else {
            state.eoePlayerProfile = {
              kind:"PC", attributes:sheet.attributes, skills:sheet.skills,
              specialties:setup.picks.slice(), template:"10-8-7-5", schemaVersion:1
            };
            setup.phase = "complete";
            eoeEnsurePlayerProfileCard();
            eoeEnsureCharacterHealth();
            response = {ok:true, complete:true, reason:`${rank}: ${name}. Character setup complete.`};
            log(`SKL SETUP | complete specialties=${setup.picks.join(" / ")}`);
          }
        } else {
          setup.phase = ["primary", "secondary", "tertiary"][setup.picks.length];
          response = {ok:true, complete:false, reason:`${rank}: ${name} saved.`};
          log(`SKL SETUP | ${rank}=${name}`);
        }
      }
    }
  }
  setup.lastActionId = actionId;
  setup.lastResponse = response;
  if (!response.ok) log(`SKL SETUP REJECT | ${response.reason}`);
  return response;
};

/** Create one new vessel when assigned for an outbound expedition. The
 * first assignment happens during pre-launch preparation; later assignments
 * happen at the next launch. Both random rolls are saved to persistent state.
 */
const eoeRollExpeditionShip = () => {
  if (!state.eoeShipHistory || !Array.isArray(state.eoeShipHistory)) state.eoeShipHistory = [];
  if (!Number.isInteger(state.eoeNextShipId) || state.eoeNextShipId < 1) state.eoeNextShipId = 1;
  const serial = state.eoeNextShipId++;
  const id = `voyage_${String(serial).padStart(3, "0")}`;
  const used = new Set(state.eoeShipHistory.map(s => s.name));
  if (state.eoeCurrentShip) used.add(state.eoeCurrentShip.name);
  let nameRoll = 1, epithetRoll = 1, name = "";
  for (let attempt = 0; attempt < 30; attempt++) {
    nameRoll = Math.floor(Math.random() * 100) + 1;
    epithetRoll = Math.floor(Math.random() * 20) + 1;
    const stem = EOE_SHIP_NAME_FIRST[Math.floor((nameRoll - 1) / 10)];
    const tail = EOE_SHIP_NAME_LAST[(nameRoll - 1) % 10];
    name = `The ${EOE_SHIP_EPITHETS[epithetRoll - 1]} ${stem} ${tail}`;
    if (!used.has(name)) break;
  }
  // Guarantee unique names even if the crew somehow uses all 2,000 options.
  if (used.has(name)) name += ` ${serial}`;
  return {
    id, name, nameRoll, epithetRoll,
    designation: 'EIR-03 "Triad"', passengerCapacity: 3,
    cargoSlots: 150, nominalFreeCargoSlots: 96,
    status: "ASSIGNED"
  };
};

const eoeShipCardIndex = ship => {
  if (!ship || typeof storyCards === "undefined" || !Array.isArray(storyCards)) return -1;
  const marker = eoeShipRecordMarker(ship);
  return storyCards.findIndex(c => String(c.entry || "").includes(marker));
};
const eoeEnsureShipCard = ship => {
  if (!ship) return false;
  const keys = eoeShipKey(ship);
  const body = eoeShipCardText(ship);
  const idx = eoeShipCardIndex(ship);
  if (idx >= 0) {
    const c = storyCards[idx];
    if (c.keys !== keys || c.entry !== body || c.type !== EOE_SHIP_CARD_TYPE)
      updateStoryCard(idx, keys, body, EOE_SHIP_CARD_TYPE, ship.name);
    return true;
  }
  // Refuse to overwrite another author's Story Card, even if its keys clash.
  if (typeof storyCards !== "undefined" && Array.isArray(storyCards) &&
      storyCards.some(c => c.keys === keys)) {
    log(`SHIP CARD | key collision for ${ship.id}; unrelated card preserved`);
    return false;
  }
  const added = addStoryCard(keys, body, EOE_SHIP_CARD_TYPE, ship.name);
  if (added === false) {
    log(`SHIP CARD | add failed for ${ship.id}`);
    return false;
  }
  log(`SHIP CARD | created id=${ship.id} name=${ship.name}`);
  return true;
};
const eoeRetireShipCard = ship => {
  if (!ship) return;
  const idx = eoeShipCardIndex(ship);
  if (idx < 0) return;
  const archivedKey = `eoe_archived_ship_${ship.id}`; // not normal prose
  const old = storyCards[idx];
  if (old.type === EOE_SHIP_ARCHIVED_TYPE && old.keys === archivedKey) return;
  updateStoryCard(idx, archivedKey,
    `${eoeShipRecordMarker(ship)}\nFormer expedition vessel: ${ship.name}.\nNo longer assigned to the active crew.\nArchived record; not a current ship description.`,
    EOE_SHIP_ARCHIVED_TYPE, `FORMER SHIP — ${ship.name}`);
  log(`SHIP CARD | archived id=${ship.id}`);
};
const eoeAssignShipToOutboundExpedition = () => {
  const prior = state.eoeCurrentShip || null;
  const ship = eoeRollExpeditionShip();
  if (!eoeEnsureShipCard(ship)) {
    // Never silently switch ship state if the card could not be created.
    state.eoeNextShipId--;
    return null;
  }
  if (prior) {
    eoeRetireShipCard(prior);
    state.eoeShipHistory.push({ id:prior.id, name:prior.name, designation:prior.designation });
  }
  state.eoeCurrentShip = ship;
  log(`SHIP ASSIGNED | ${ship.id} ${ship.name} type=${ship.designation} nameD100=${ship.nameRoll} adjD20=${ship.epithetRoll}`);
  return ship;
};

/** Increment once per completed OUTPUT response/action, never per hook.

  if (!state.atdwTurnReminders) state.atdwTurnReminders = {};
  const id = typeof info !== "undefined" && Number.isFinite(info.actionCount)
    ? info.actionCount : null;
  state.atdwTurnReminders[name] = { count: 0, lastAction: id };
};
const atdwTickTurnReminder = (name, interval) => {
  if (!state.atdwTurnReminders) state.atdwTurnReminders = {};
  if (!state.atdwTurnReminders[name]) atdwResetTurnReminder(name);
  const r = state.atdwTurnReminders[name];
  const id = typeof info !== "undefined" && Number.isFinite(info.actionCount)
    ? info.actionCount : null;
  if (id != null && id === r.lastAction) return false;
  if (id != null) r.lastAction = id;
  r.count++;
  return r.count > 0 && r.count % Math.max(1, interval || 10) === 0;
};

/** Fail fast if any generated area becomes structurally orphaned.
 * Locked/blocked doors are a separate *operational* accessibility policy.
 */
const atdwValidateSiteGraph = (site) => {
  const keys = Object.keys(site.nodes || {});
  if (!site.nodes["0"]) throw Error("Graph lacks Node 0");
  const seen = new Set(["0"]), queue=["0"];
  for(let head=0; head<queue.length; head++) {
    const node=site.nodes[queue[head]];
    for(const e of node.exits || []) {
      const k=String(e.to);
      if (!site.nodes[k] || !site.passages[e.passageId]) throw Error(`Invalid graph edge ${node.areaId}->${k}`);
      const back=site.nodes[k].exits.some(r => r.passageId===e.passageId && String(r.to)===String(node.areaId));
      if(!back) throw Error(`No reverse graph edge ${node.areaId}->${k}`);
      if(!seen.has(k)) {seen.add(k);queue.push(k);}
    }
  }
  if(seen.size!==keys.length) throw Error(`Orphan graph nodes: ${keys.filter(k=>!seen.has(k)).join(",")}`);
  return true;
};

const randomEntry = (table) =>

  table[Math.floor(Math.random() * table.length)];

const getPlaceholder = (question) => {

  const placeholders = state.placeholders || [];

  return placeholders.find(p => p.question === question)?.answer?.trim() || "";

};

/**
 * CREATURE CARD GENERATOR
 * Rolls a creature using the original tables. With createCard=false, stores
 * the creature's properties but delays its Story Card until first discovery.
 * This prevents undiscovered Site creatures from filling story-card context.
 */
// CREATURE CARD GENERATOR

// ============================================================

const generateCreatureCard = (createCard = true) => {

  // ----------------------------------------------------------

  // RANDOM CREATURE DETAILS

  // ----------------------------------------------------------

  const creatureName =

    generateThreePartName();

  const creatureClass =

    randomEntry(creatureClasses);

  const creatureTraitValue =

    randomEntry(creatureTrait);

  const creatureAbilityValue =

    randomEntry(creatureAbility);

  const creatureDriveValue =

    randomEntry(creatureDrive);

  const creatureAppearanceValue =

    randomEntry(creatureAppearance);

  const creatureCoveredValue =

    randomEntry(creatureCovered);

  const creatureLimbsValue =

    randomEntry(creatureLimbs);

  const creatureLimbTypeValue =

    randomEntry(creatureLimbtype);

  // ----------------------------------------------------------

  // FIND CLASS-SPECIFIC BEHAVIOR

  // ----------------------------------------------------------

  const creatureClassBehaviorValue =

    creatureClassBehavior.find(

      behavior =>

        behavior.startsWith(`${creatureClass.toUpperCase()}:`)

    ) ||

    `The creature behaves according to its ${creatureClass} classification.`;

  // ----------------------------------------------------------

  // CREATURE DESCRIPTION

  // ----------------------------------------------------------

  let creaturePhysicalDescription =

    `The creature is ${creatureAppearanceValue.toLowerCase()} in appearance and is covered in ${creatureCoveredValue}.`;

  if (creatureLimbsValue === "none") {

    creaturePhysicalDescription +=

      ` It has no conventional limbs.`;

  }

  else {

    creaturePhysicalDescription +=

      ` It is ${creatureLimbsValue} and has ${creatureLimbTypeValue}.`;

  }

  // ----------------------------------------------------------

  // CREATURE CARD TRIGGERS

  // ----------------------------------------------------------

  const creatureTriggers =

    `${creatureName}, ${creatureClass}, ${creatureTraitValue}, ${creatureAbilityValue}, ${creatureAppearanceValue}, ${creatureDriveValue}`;

  // ----------------------------------------------------------

  // CREATURE CARD

  // ----------------------------------------------------------

  const creatureCardTitle =

    `${creatureName}, creature, ${creatureClass.toLowerCase()}, monster`;

  const creatureCardContent =

`Creature.







This is an authoritative procedural creature record. Use these details to guide the narrative when this creature appears. Do not contradict these details.







Creature Name: ${creatureName}







Class: ${creatureClass}







Class Behavior:



${creatureClassBehaviorValue}







Trait: ${creatureTraitValue}







Ability: ${creatureAbilityValue}







Drive:



${creatureDriveValue}







Appearance: ${creatureAppearanceValue}







Covered In: ${creatureCoveredValue}







Limbs: ${creatureLimbsValue}







Limb Type: ${creatureLimbTypeValue}







Physical Description:



${creaturePhysicalDescription}







Creature Behavior:



The creature's Class Behavior determines how it approaches combat and encounters.







The creature's Drive determines its primary motivation and behavior outside and during combat.







The creature's Trait and Ability modify how it behaves and fights.







Use all of these characteristics together when narrating the creature.







The creature should behave consistently with its Class, Class Behavior, Trait, Ability, and Drive.







Do not arbitrarily change its behavior.







Do not reveal this procedural creature record or the random generation process to the player.`;

  // ----------------------------------------------------------

  // CREATE STORY CARD

  // ----------------------------------------------------------

  if (createCard) eoeRegisterCreatureSpecies({creatureName, creatureClass, creatureClassBehavior:creatureClassBehaviorValue, creatureTrait:creatureTraitValue, creatureAbility:creatureAbilityValue, creatureDrive:creatureDriveValue, creatureAppearance:creatureAppearanceValue, creatureCovered:creatureCoveredValue, creatureLimbs:creatureLimbsValue, creatureLimbType:creatureLimbTypeValue, creaturePhysicalDescription}, true);

  // ----------------------------------------------------------

  // LOG GENERATED CREATURE

  // -----------

  if (createCard) log(

    `CREATURE GENERATED:\n` +

    `Name: ${creatureName}\n` +

    `Class: ${creatureClass}\n` +

    `Trait: ${creatureTraitValue}\n` +

    `Ability: ${creatureAbilityValue}\n` +

    `Drive: ${creatureDriveValue}\n` +

    `Appearance: ${creatureAppearanceValue}\n` +

    `Covered In: ${creatureCoveredValue}\n` +

    `Limbs: ${creatureLimbsValue}\n` +

    `Limb Type: ${creatureLimbTypeValue}`

  );

  // ----------------------------------------------------------

  // RETURN CREATURE DATA

  // ----------------------------------------------------------

  return {
    cardCreated: createCard,
    creatureName: creatureName,
    creatureClass: creatureClass,
    creatureClassBehavior: creatureClassBehaviorValue,
    creatureTrait: creatureTraitValue,
    creatureAbility: creatureAbilityValue,
    creatureDrive: creatureDriveValue,
    creatureAppearance: creatureAppearanceValue,
    creatureCovered: creatureCoveredValue,
    creatureLimbs: creatureLimbsValue,
    creatureLimbType: creatureLimbTypeValue,
    creaturePhysicalDescription: creaturePhysicalDescription
  };

};

// CBT-01D6: A creature's generated word names its SPECIES, not its individual.
// One canonical phenotype is kept in persistent state; specimens retain unique
// health/encounter IDs and can differ in wounds, posture, and location.
const EOE_SPECIES_CARD_TYPE = "EOE_CREATURE_SPECIES";
const EOE_SPECIES_FIELDS = ["creatureClass", "creatureClassBehavior", "creatureTrait",
  "creatureAbility", "creatureDrive", "creatureAppearance", "creatureCovered",
  "creatureLimbs", "creatureLimbType", "creaturePhysicalDescription"];
const eoeSpeciesKey = name => String(name || "").trim().toLowerCase();
const eoeSpeciesStore = () => {
  if(state.eoeSpeciesRegistry === undefined)
    state.eoeSpeciesRegistry = {schemaVersion:1,records:{}};
  const r=state.eoeSpeciesRegistry;
  return r && r.schemaVersion===1 && r.records &&
    typeof r.records==='object' && !Array.isArray(r.records) ? r : null;
};
const eoeSpeciesCardBody = record =>
  `CREATURE SPECIES — ${record.name}\n`+
  `This is a species or kind, NOT an individual creature's personal name. `+
  `Use "a ${record.name.toLowerCase()}" and "the ${record.name.toLowerCase()}". `+
  `Separate specimens may have different health, locations and fates.\n`+
  `Appearance: ${record.creaturePhysicalDescription}\n`+
  `Classification: ${record.creatureClass}. Trait: ${record.creatureTrait}. `+
  `Ability: ${record.creatureAbility}.\n`+
  `Typical drive: ${record.creatureDrive}. Behavior: ${record.creatureClassBehavior}\n`+
  `Describe this species consistently. This entry records typical biology and `+
  `behavior, not that any specific specimen is present or alive. `+
  `JavaScript owns individual Wounds, locations, and deaths.`;
const eoeSpeciesCardEnsure = record => {
  if(!record || typeof storyCards==='undefined' || !Array.isArray(storyCards)) return false;
  const name=record.name;
  const keys=`${name}, ${name.toLowerCase()}s`;
  const old=storyCards.findIndex(c=>c &&
    (String(c.name||c.title||'').toLowerCase()===name.toLowerCase() ||
      String(c.keys||'').split(',')[0].trim().toLowerCase()===name.toLowerCase()) &&
    (c.type===EOE_SPECIES_CARD_TYPE ||
     (c.type==='Creature' && /authoritative procedural creature record/i.test(String(c.entry||c.value||'')))));
  const body=eoeSpeciesCardBody(record);
  if(old>=0){
    const c=storyCards[old];
    if(c.type===EOE_SPECIES_CARD_TYPE && String(c.entry||c.value||'')===body) return true;
    if(typeof updateStoryCard!=='function') return false;
    updateStoryCard(old,keys,body,EOE_SPECIES_CARD_TYPE,name);
    log(`SPECIES CARD UPDATED | species=${name}`);
    return true;
  }
  // Never modify an unrelated Story Card merely because its triggers overlap.
  if(storyCards.some(c=>c && c.keys===keys)){
    log(`SPECIES CARD BLOCKED | species=${name} key collision`);
    return false;
  }
  if(typeof addStoryCard!=='function') return false;
  const index=addStoryCard(keys,body,EOE_SPECIES_CARD_TYPE,name);
  if(index===false || index===null || index===undefined){
    log(`SPECIES CARD BLOCKED | species=${name}`);
    return false;
  }
  log(`SPECIES CARD CREATED | species=${name}`);
  return true;
};
const eoeRegisterCreatureSpecies = (creature, makeCard=true) => {
  if(!creature || typeof creature.creatureName!=='string' ||
     !/^[a-z][a-z0-9 -]{1,63}$/i.test(creature.creatureName.trim())) return null;
  const store=eoeSpeciesStore();if(!store)return null;
  const key=eoeSpeciesKey(creature.creatureName);
  let record=store.records[key];
  if(record){
    if(record.schemaVersion!==1 || typeof record.name!=='string' || record.name.toLowerCase()!==key) return null;
    // A generated name reused by coincidence must not redefine its species.
    for(const field of EOE_SPECIES_FIELDS)
      if(typeof record[field]==='string') creature[field]=record[field];
  }else{
    if(Object.keys(store.records).length>=128){
      log(`SPECIES REGISTRY FULL | species=${creature.creatureName}`);
      return null;
    }
    record={schemaVersion:1,name:creature.creatureName.trim()};
    for(const field of EOE_SPECIES_FIELDS) record[field]=String(creature[field]||'').slice(0,1100);
    store.records[key]=record;
    log(`SPECIES REGISTERED | species=${record.name}`);
  }
  if(makeCard) eoeSpeciesCardEnsure(record);
  return record;
};

// Upgrade visible legacy encounters only. Never enumerate undiscovered Site
// creatures into Story Cards and never revive creatures that were killed.
const eoeEnsureVisibleCreatureSpecies = () => {
  const ensure=c=>{
    if(!c || !c.creatureName) return;
    const store=eoeSpeciesStore();if(!store)return;
    const record=store.records[eoeSpeciesKey(c.creatureName)];
    // Do not rewrite a living specimen's situational Drive/behavior every
    // turn: a species describes a tendency, while an individual may adapt.
    if(record) eoeSpeciesCardEnsure(record);
    else eoeRegisterCreatureSpecies(c,true);
  };
  const active=eoeDebugCreatureAtLocation();
  if(active && active.entry && active.entry.status==='present') ensure(active.creature);
  const site=state.currentSiteId && state.sites && state.sites[state.currentSiteId];
  if(!site || site.mode!=='EXPLORING_ACTIVE' || !site.nodes) return;
  const node=site.nodes[String(site.currentAreaId)];
  if(node && node.visited && node.encounterTriggered && node.encounterCreature)
    ensure(node.encounterCreature);
};

// ============================================================================
// SKL-01B2 — AUTHORITATIVE NPC ATTRIBUTE PROFILES
// ATDW NPCs have no 12-skill point pool: they resolve a skill using its
// associated ATTRIBUTE (the lookup exists in SKL-01A above).
// Each NPC receives six starting attributes totaling 60. Existing narrative
// skills are only a role hint, never treated as numeric scores. No new rolls.
// These profiles survive turns, travel, and revisits; Story Cards are views.
// ============================================================================
const EOE_NPC_PROFILE_VERSION = 1;
const EOE_NPC_FOCUS_CUES = [
  {attr: "INT",  re: /\b(?:engineer(?:ing)?|mechanic(?:al)?|technician|technolog(?:y|ist)|electronic(?:s)?|computers?|hacking|hacker|eirseid|research(?:er)?|scien(?:ce|tist)|analysis|analyst|programmer|repair)\b/i},
  {attr: "DEX",  re: /\b(?:medic(?:ine|al)?|first aid|doctor|surgeon|nurs(?:e|ing)|pilot(?:ing)?|flier|flyer|stealth|sneak|ranged|shoot(?:er|ing)?|snip(?:e|er)|marksmanship|driv(?:e|er|ing))\b/i},
  {attr: "STR",  re: /\b(?:close combat|melee|brawl(?:er|ing)?|wrestl(?:e|er|ing)|athlet(?:e|ics)|strong|lifting|bodyguard|heavy labor)\b/i},
  {attr: "WIL",  re: /\b(?:surviv(?:al|or)|tracking|tracker|perception|scout(?:ing)?|recon(?:naissance)?|resolv(?:e|ed)|endurance|pathfind(?:er|ing))\b/i},
  {attr: "CHA",  re: /\b(?:manipulat(?:ion|e|or)|negotiat(?:e|ion|or)|diploma(?:cy|t)|persuas(?:ion|ive)|leader(?:ship)?|merchant|trading|trader|social)\b/i}
];
// Focus: +2 to a matching role attribute, -2 to a counterbalance, with four
// neutral scores. It is one permitted distribution of the rulebook's 60 pts.
const EOE_NPC_WEAK_ATTRIBUTE = {INT:"CHA", DEX:"CHA", STR:"INT", WIL:"STR", CHA:"STR"};
const eoeNpcAttributesFromNarrative = rawSkills => {
  const attributes = {STR:10, DEX:10, CON:10, WIL:10, INT:10, CHA:10};
  const skills = String(rawSkills == null ? "" : rawSkills).slice(0, 1200);
  const hits = EOE_NPC_FOCUS_CUES.filter(c => c.re.test(skills));
  // Two different areas of expertise: choose the first-mentioned cue; in a
  // tie prefer higher-level rulebook specificity, never depend on the NPC name.
  hits.sort((a,b) => skills.search(a.re)-skills.search(b.re));
  const focus = hits.length ? hits[0].attr : null;
  if (focus) {
    attributes[focus] += 2;
    attributes[EOE_NPC_WEAK_ATTRIBUTE[focus]] -= 2;
  }
  return {attributes, focus};
};
const eoeNewCrewNpcProfile = (slot, name, narrativeSkills) => {
  if (![2,3].includes(slot) || typeof name !== "string" || !name.trim()) return null;
  const built = eoeNpcAttributesFromNarrative(narrativeSkills);
  return {
    kind:"NPC", slot, name:name.trim(), attributes:built.attributes,
    focus:built.focus, schemaVersion:EOE_NPC_PROFILE_VERSION
  };
};
const eoeNewArchetypeCrewProfile = (slot,name,choice) => {
  if (![2,3].includes(slot) || !String(name || "").trim() ||
      !choice || !choice.archetype) return null;
  const attributes=eoeNpcAttributesFromArchetype(choice.archetype);
  if (!attributes) return null;
  return {kind:"NPC",slot,name:String(name).trim(),attributes,
    archetype:choice.archetype,archetypeSelection:choice.selected,
    focus:null,schemaVersion:2};
};

const eoeNpcAttributeBlock = profile => {
  const scores = EOE_ATTRIBUTE_NAMES.map(a=>`${a} ${profile.attributes[a]}`).join(" | ");
  const slot = profile.slot;
  return `[EOE NPC STATS SLOT ${slot} START]\n` +
    (profile.archetype ? `Expedition archetype: ${profile.archetype}\n` +
      `Role strengths: ${EOE_CREW_ARCHETYPE_DESCRIPTIONS[profile.archetype]}\n` : "") +
    `Attributes: ${scores}\n` +
    `NPC skill checks use linked attributes (not a numerical skill pool).\n` +
    `${eoeHealthDisplay(`crew:${slot}`)}\n` +
    `JavaScript profiles and Wound counts are authoritative.\n` +
    `[EOE NPC STATS SLOT ${slot} END]`;
};
/**
 * Update only a recognized EoE-generated crew card. Protect author cards
 * with coincidentally matching keys, and keep manual prose added elsewhere.
 */
const eoeEnsureCrewNpcCard = profile => {
  if (!profile || typeof storyCards === "undefined" || !Array.isArray(storyCards)) return false;
  const key = `${profile.name}, crew`;
  const begin = `[EOE NPC STATS SLOT ${profile.slot} START]`;
  const end = `[EOE NPC STATS SLOT ${profile.slot} END]`;
  const marked = storyCards.map((c,i)=>({c,i})).filter(({c}) =>
    c.type === "Character" && String(c.entry||"").includes(begin));
  if (marked.length > 1) {log(`SKL NPC CARD | slot=${profile.slot} multiple marked cards; deferred`);return false;}
  let idx;
  if (marked.length === 1) {
    idx = marked[0].i;
    if (storyCards[idx].keys !== key) {
      log(`SKL NPC CARD | slot=${profile.slot} name/key conflict; deferred`);
      return false;
    }
  } else {
    const matching = storyCards.map((c,i)=>({c,i})).filter(({c})=>c.keys === key);
    if (matching.length !== 1 || matching[0].c.type !== "Character" ||
        !String(matching[0].c.entry || "").startsWith("Pathfinder crew member.")) {
      log(`SKL NPC CARD | slot=${profile.slot} missing or unmanaged card; deferred`);
      return false;
    }
    idx=matching[0].i;
  }
  const card=storyCards[idx];
  const current=String(card.entry||"");
  const block=eoeNpcAttributeBlock(profile);
  const start=current.indexOf(begin);
  let next;
  if (start>=0) {
    const stop=current.indexOf(end,start);
    if (stop<0) {log(`SKL NPC CARD | slot=${profile.slot} broken managed block; deferred`);return false;}
    next=current.slice(0,start)+block+current.slice(stop+end.length);
  } else {
    if (current.includes("[EOE NPC STATS SLOT ")) {
      log(`SKL NPC CARD | slot=${profile.slot} conflicting managed slot; deferred`);return false;
    }
    next=current+"\n\n"+block;
  }
  if (next===current) return true;
  if (typeof updateStoryCard!=="function") {
    log(`SKL NPC CARD | slot=${profile.slot} update unavailable; deferred`);return false;
  }
  updateStoryCard(idx,card.keys,next,card.type,card.name);
  log(`SKL NPC CARD | slot=${profile.slot} updated name=${profile.name}`);
  return true;
};
/**
 * Called during Input initialization, after the existing crew-card generator.
 * Profile generation is once per slot; retrying updates missing Story Card
 * projections but never rerolls or reinterprets established attributes.
 */
const eoeEnsureCrewNpcProfiles = () => {
  if (!state.eoeCrewProfiles || typeof state.eoeCrewProfiles !== "object" ||
      Array.isArray(state.eoeCrewProfiles)) state.eoeCrewProfiles = {};
  // New adventures defer numeric NPC registration until two actual choices.
  // During explicit reconfiguration, keep existing stats untouched until
  // both validated selections are committed as a single operation.
  if (state.eoeCrewArchetypeSetup && state.eoeCrewArchetypeSetup.phase !== "complete") return;
  const names = [getPlaceholder("crew #2 name"), getPlaceholder("crew #3 name")];
  if (names[0] && names[1] && String(names[0]).trim()===String(names[1]).trim()) {
    log("SKL NPC | duplicate crew names; profiles deferred");return;
  }
  [2,3].forEach((slot,i)=>{
    const name=String(names[i] || "").trim();
    if (!name) return; // Missing placeholder may populate on another turn.
    let profile=state.eoeCrewProfiles[slot];
    if (profile && profile.name!==name) {
      log(`SKL NPC | slot=${slot} persistent name conflict; preserved`);return;
    }
    const choice=state.eoeCrewArchetypes && state.eoeCrewArchetypes[slot];
    if (choice && (!profile || profile.archetype !== choice.archetype ||
        profile.archetypeSelection !== choice.selected)) {
      // Deliberate #CREWSETUP may replace attributes, never biography.
      const updated=eoeNewArchetypeCrewProfile(slot,name,choice);
      if (!updated) return;
      profile=updated;
      state.eoeCrewProfiles[slot]=profile;
      log(`SKL NPC | slot=${slot} archetype=${profile.archetype} selected=${profile.archetypeSelection}`);
    } else if (!profile) {
      const skills=getPlaceholder(`crew #${slot} skills`);
      profile=eoeNewCrewNpcProfile(slot,name,skills);
      if (!profile) return;
      state.eoeCrewProfiles[slot]=profile;
      log(`SKL NPC | slot=${slot} registered name=${name} focus=${profile.focus || "balanced"}`);
    }
    eoeEnsureCrewNpcCard(profile);
  });
};

/**

 * pending and retry on later INPUT turns rather than permanently losing it.

// HP-01C4 — FIRST AUTHORITATIVE TRAVEL HAZARD
// Called only by INPUT when #CONTINUE reaches the scheduled event day.
// A tool falls from its shelf and inflicts exactly one Wound on a uniformly
// selected living registered party member. All other travel hazards retain
// narrative-only/deferred effects; never infer damage from free-text prose.
// The event's identity comes from the ACTION in which the checkpoint begins
// and is also cached on the voyage. This prevents retries and new narration
// from applying a second Wound. No action identity => no untraceable damage.
// ============================================================================
const eoeResolveToolFallHazard = (travel, actionId) => {
  const failure = error => ({ok:false,error});
  if (!travel || travel.event!==EOE_TOOL_FALL_EVENT ||
      travel.phase!=="event" || travel.eventTriggered!==true)
    return failure("No tool-fall travel event is active.");
  if (travel.eoeToolFallResolution !== undefined)
    return travel.eoeToolFallResolution;
  if (!Number.isSafeInteger(actionId) || actionId<0)
    return failure("No stable action identity; no Wound applied.");
  // Fail closed on invalid or future health schemas. No fallback capacity.
  const health=eoeHealthStore();
  if (!health) return failure("Health store unavailable; no Wound applied.");
  const ids=["pc","crew:2","crew:3"].filter(id => {
    const r=eoeHealthRecord(id);
    return eoeHealthValidWoundRecord(r) &&
      (id==="pc" ? r.kind==="PC" : r.kind==="NPC") &&
      r.status!=="dead" && r.woundsRemaining>0;
  });
  if (!ids.length) return failure("No living registered party member can be injured.");
  const roll=Math.random();
  const chosen=ids[Math.min(ids.length-1,Math.floor(roll*ids.length))];
  const wound=eoeHealthApplyWounds({
    id:chosen,wounds:1,cause:"falling unsecured maintenance tool",
    eventId:`travel:tool-fall:action:${actionId}`
  });
  if (!wound.ok) {
    log(`TRAVEL TOOL FALL BLOCKED | ${wound.error}`);
    return failure(wound.error);
  }
  const record=eoeHealthRecord(chosen);
  const result={ok:true,characterId:chosen,
    characterName:chosen==="pc"?"Player":record.name,
    actionId,remaining:wound.remaining,fatal:!!wound.fatal,
    eventId:`travel:tool-fall:action:${actionId}`,applied:wound.applied};
  travel.eoeToolFallResolution=result;
  log(`TRAVEL TOOL FALL | target=${chosen} remaining=${wound.remaining}/${record.maxWounds} action=${actionId}`);
  return result;
};

  // Favor rooms appropriate to the Site, without making every room identical.
  const chosen = priority && Math.random() < 0.6
    ? roomSceneTemplates[priority[Math.floor(Math.random() * priority.length)]]
    : randomEntry(roomSceneTemplates);
  const scale = randomEntry(roomSceneScales);
  const surface = randomEntry(roomSceneSurfaces);
  return `The chamber is a ${scale} ${chosen[0]} containing ${chosen[1]}. ` +

/** Narrate only what is genuinely present in the current generated node. */
const describeRoomContents = (node, isFirstVisit) => {
  const c = node.encounterCreature;
  const artifact = node.xenoanthropologicalArtifact;
  const hazard = node.hazardManifestation;
  const past = !isFirstVisit;
  if (node.result === "Encounter" && c) {
    const physical = String(c.creaturePhysicalDescription || `${c.creatureAppearance || "unfamiliar"} form`).replace(/\s+/g, " ").trim();
    const appearance = String(c.creatureAppearance || "unfamiliar").toLowerCase();
    const creatureClass = String(c.creatureClass || "creature").toLowerCase();
    const article = /^[aeiou]/.test(appearance) ? "an" : "a";
    const drive = String(c.creatureDrive || "its own motives").replace(/\s+/g, " ").trim();
    return past
      ? `The crew previously encountered a ${c.creatureName.toLowerCase()} here. ${physical} Its present fate follows the story already told; do not reintroduce it or restore it if defeated.`
      : `Within this space is a ${c.creatureName.toLowerCase()}, ${article} ${appearance} ${creatureClass}. ${physical} It exhibits ${c.creatureTrait || "unusual traits"} and ${c.creatureAbility || "unusual abilities"}; its behavior is driven by ${drive}.`;
  }
  if (node.result === "Xenoanthropological Artifact" && artifact) {
    return past
      ? `During the first survey, the crew discovered ${artifact.description} here. Its current whereabouts depend on what the party subsequently did.`
      : `Among the debris lies ${artifact.description}, an unfamiliar cultural object whose purpose and value are not yet understood.`;
  }
  if (node.result === "Hazard Manifestation" && hazard) {
    const trap=node.eoeMechanicalTrap;
    if (hazard===EOE_MECHANICAL_TRAP_TEXT && trap && trap.schemaVersion===1) {
      return trap.status==="pending"
        ? "An unstable mechanism threatens anyone attempting to handle it. It has not yet struck; the player may choose #TRAP EVADE, #TRAP DISARM, or move away through an existing exit."
        : "The mechanical trap has already been resolved; do not trigger new injury here.";
    }
    return past
      ? `The room has already been associated with the danger described as: ${hazard}. Continue previously established consequences rather than triggering it anew.`
      : `An immediate complication threatens the crew: ${hazard}. Introduce its tangible cause, leaving the characters room to react.`;
  }
  // A specific discovery is rolled once and stored as prose, never a label.
  // Revisit framing includes its identity without reactivating the incident.
  if (node.discoveryScene) {
    return past
      ? `This place is already known to the crew. Its previously established discovery is: ${node.discoveryScene} Preserve the situation and any consequences already established in the story; do not reset or trigger it again. Its present condition depends on subsequent events.`
      : `${node.discoveryScene} Introduce this visible situation now, but leave any investigation, confrontation or resolution to the characters.`;
  }
  // An older Site may have been visited before the concrete narrative tables
  // existed. Never introduce a newly randomized discovery into such a room.
  if (past) return `This place is already known to the crew. Preserve its established discoveries and consequences rather than generating a fresh incident.`;
  const fallback = {
    "Resources": "Signs of recoverable supplies remain among the room's existing equipment, available for the crew to examine.",
    "Locked Access Point": "A sealed internal container or compartment requires investigation; it is not a new graph exit.",
    "Dread Event": "An unsettling environmental anomaly makes its presence felt in the chamber.",
    "Automated Security Measures": "An old security mechanism monitors the chamber, demanding identification before granting access to internal equipment.",
    "Fixed Event": "A piece of damaged equipment presents an immediate problem that may require the crew's attention.",
    "Random Occurrence": "A dormant machine unexpectedly starts an unfamiliar routine within the chamber."
  };
  return fallback[node.result] || "The party can examine the space and discover its details without assuming an outcome.";
};

/**
 * Narrate the actual passage and destination together. The same roomScene
 * is reused forever; only the connector and first/revisit framing changes.
 * Unknown legacy visited rooms get a conservative generic anchor, never a
 * fresh random room type that could contradict established story history.
 */
const composeGraphSceneNarrative = (site, node, last, firstVisit) => {
  const moving = !!(last && last.to === node.areaId && last.needsNarration);
  let opening = "";
  if (moving && last.from != null) {
    const origin = getSiteNode(site, last.from);
    const chosen = origin && origin.exits[last.exitNumber - 1];
    const path = chosen ? chosen.description : "A previously mapped passage connects these areas.";
    opening = firstVisit
      ? `The route into this area follows a passage: ${path} `
      : `The crew returns along a familiar passage: ${path} `;
  } else if (moving) {
    opening = "Beyond the Site entrance, the crew reaches its first interior area. ";
  } else {
    opening = `The crew remains in this same area. `;
  }
  const identity = node.roomScene ||
    (node.visited
      ? "This chamber has already been described in the story. Its established function, layout, furnishings and damage remain exactly as the crew previously found them; do not invent a replacement room type."
      : "An unclassified interior chamber lies ahead; preserve any details already established for it.");
  const contents = describeRoomContents(node, firstVisit);
  // Persisted background detail, never a second generated obstacle, object, or exit.
  const evidence = node.backgroundEvidence
    ? `An incidental environmental clue already present here is: ${node.backgroundEvidence} ` +
      `This clue is descriptive only; it cannot invent a second rolled discovery or a new route.`
    : "";
  return `${opening}${identity} ${contents} ${evidence}`;
};

/**
 * COMPACT SITE BLUEPRINT LOGGER
 * Legacy blueprint diagnostics; the graph engine adds a capped topology digest for testing.
 */
// COMPACT SITE BLUEPRINT LOGGER

// Maximum line width: 40 characters

// ============================================================

const logSiteBlueprint = (siteId) => {

  const site = state.sites?.[siteId];

  if (!site) {

    log(`SITE NOT FOUND: ${siteId}`);

    return;

  }

  const WIDTH = 40;

  // Wrap long text without losing information.

  const wrap = (label, value) => {

    const words = String(value ?? `None`)

      .replace(/\s+/g, ` `)

      .split(` `);

    const lines = [];

    let line = label;

    for (const word of words) {

      if ((line + word).length > WIDTH) {

        lines.push(line.trimEnd());

        line = `   `;

      }

      // Split exceptionally long words too.

      let remaining = word;

      while (remaining.length > WIDTH - 3) {

        if (line.trim()) {

          lines.push(line.trimEnd());

        }

        lines.push(`   ` + remaining.slice(0, WIDTH - 3));

        remaining = remaining.slice(WIDTH - 3);

        line = `   `;

      }

      line += remaining + ` `;

    }

    if (line.trim()) {

      lines.push(line.trimEnd());

    }

    return lines;

  };

  const explored = site.explored || [];

  const unexplored = site.unexplored || [];

  const areas = [...explored, ...unexplored]

    .sort((a, b) => a.areaId - b.areaId);

  const lines = [
    `SITE: ${site.name}`,
    `ID: ${site.siteId}`,
    `SIZE: ${site.size}`,
    `AREAS: ${areas.length}`,
    `DONE: ${explored.length}`,
    `LEFT: ${unexplored.length}`,
    `----------------------------------------`
  ];

  for (const area of areas) {

    const status = explored.some(

      a => a.areaId === area.areaId

    ) ? `E` : `U`;

    lines.push(...wrap(

      `${String(area.areaId).padStart(2, `0`)} ${status} | `,

      area.result

    ));

    lines.push(...wrap(

      `   Path: `,

      area.connector

    ));

    if (area.ambush) {

      lines.push(`   Ambush: Yes`);

    }

    if (area.encounterCreature) {

      lines.push(...wrap(

        `   Creature: `,

        area.encounterCreature.creatureClass ||

        `Generated creature`

      ));

    }

    if (area.hazardManifestation) {

      lines.push(...wrap(

        `   Hazard: `,

        area.hazardManifestation

      ));

    }

    if (area.xenoanthropologicalArtifact) {

      lines.push(...wrap(

        `   Artifact: `,

        area.xenoanthropologicalArtifact.description

      ));

    }

  }

  lines.push(`----------------------------------------`);

  lines.push(`END SITE`);

  // One log call, with the complete digest.

  log(lines.join(`\n`));

};

/**

 *
 * The original d100 table had two non-playable outcomes:
 *   1–2   instantaneous death, 3–15 no Site discovered.
 * These are removed from V1 because an expedition should always reach
 * somewhere the crew can explore. Travel events and Site hazards still
 * present danger, but there are no unavoidable trip-ending arrival rolls.
 *
 * Retain the relative likelihoods of the four ORIGINAL usable results:
 *   20/85 derelict vessel, 25/85 planet, 20/85 station/outpost,
 *   20/85 asteroid/planetoid base.
 * A single d85 roll is used instead of rejection sampling so this always
 * terminates and consumes exactly one random draw. JS is authoritative.
 */
const rollArrival = () => {
  const roll = Math.floor(Math.random() * 85) + 1;
  let result;
  if (roll <= 20) {
    result = "The Site is a large derelict ship. The ship automatically docks in it.";
  } else if (roll <= 45) {
    result = "The crew's ship touches down on a planet.";
  } else if (roll <= 65) {
    result = "The Site is a station or outpost orbiting a planet or moon.";
  } else {
    result = "The ship docks in an Eirseid base inside an asteroid or planetoid.";
  }
  return { roll, fatal: false, noSite: false, result };
};

/**
 * GRAPH EXPLORATION ENGINE — CONSTANTS AND LABELS
 *
 * Purpose:
 *   Give every Site a persistent, connected graph of rooms and passages.
 *   A node is an Area; an outgoing edge is one player-selectable exit.
 *   Store plain objects in state so data survives AI Dungeon serialization.
 *
 * State:
 *   state.sites[siteId].nodes, .passages, .currentAreaId, .mode.
 *
 * NOTE: The original exploration and creature tables above are preserved.
 */
// ============================================================================

const graphLandmarks = [
  "amber glyphs", "copper veins", "blue lamps", "white conduits",
  "spiral etchings", "silver ribs", "cracked panels", "red markings",
  "pale crystals", "violet lights", "hanging cables", "dark mirrors",
  "frosted plates", "floating dust", "black pillars", "green sigils",
  "faint chimes", "warped lattice", "fractured tiles", "glass insets",
  "rust-colored seams", "carved spirals", "wavering light", "thin mist"
];

const graphConnectorVocabulary = (connector) => {
  const c = String(connector || "").toLowerCase();
  if (c.includes("stairs")) return ["stairs", "staircase", "steps", "climb"];
  if (c.includes("bridge")) return ["bridge", "span", "crossing", "catwalk"];
  if (c.includes("grand")) return ["wide", "grand", "hall", "hallway", "corridor"];
  if (c.includes("hallway")) return ["narrow", "hallway", "corridor", "passage"];
  return ["doorway", "threshold", "opening", "passage", "archway"];
};

const graphReverseConnector = (connector) => {
  if (connector === "Stairs going down") return "Stairs going up";
  if (connector === "Stairs going up") return "Stairs going down";
  return connector;
};

const graphConnectorDescription = (connector, landmark) => {
  const c = String(connector || "");
  if (c.includes("Stairs going down")) return `Steps descend past ${landmark} into the structure.`;
  if (c.includes("Stairs going up")) return `A staircase rises beside ${landmark}.`;
  if (c.includes("Grand")) return `A broad vaulted hallway continues past ${landmark}.`;
  if (c.includes("Simple hallway")) return `A narrow hallway bends past ${landmark}.`;
  if (c.includes("Bridge")) return `A bridge spans a hollow space beneath ${landmark}.`;
  return `An open threshold framed by ${landmark} leads onward.`;
};

/**
 * buildSiteGraph(site, blueprint)
 *
 * Input: generated Site metadata and generateSiteBlueprint()'s areas.
 * Output: a serializable, connected graph. All areas are pre-rolled once.
 * Method: build a spanning tree, then add a few optional loop passages.
 * Safety: never create isolated nodes; max four exits per node; loops and
 *   normal corridors have reverse directed edges. One-way links can be
 *   supported later without changing the navigation interface.
 */
const buildSiteGraph = (site, blueprint) => {
  const areas = Array.isArray(blueprint) ? blueprint : blueprint.areas;
  if (!Array.isArray(areas) || !areas.length) throw new Error("Site blueprint has no areas");

  const nodes = {};
  const passages = {};
  let nextPassageId = 1;

  areas.forEach(area => {
    nodes[String(area.areaId)] = {
      ...area,
      visited: false,
      visitCount: 0,
      discoveryNarrated: false,
      // V1 only records INTRODUCTION, never defeat or disposition.
      // The AI and the creature's Story Card own subsequent continuity.
      encounterTriggered: false,
      exits: []
    };
  });

  // Create two directed edges representing an ordinary two-way passage.
  const connect = (from, to, connector, ambush) => {
    const a = nodes[String(from)];
    const b = nodes[String(to)];
    const id = `p${nextPassageId++}`;
    // Distinctive landmark per exit at BOTH endpoints, avoiding ambiguous
    // keyword matches even at busy four-exit junctions.
    const availableLandmarks = graphLandmarks.filter(label =>
      !a.exits.some(e => e.landmark === label) &&
      !b.exits.some(e => e.landmark === label));
    const landmark = randomEntry(availableLandmarks.length ? availableLandmarks : graphLandmarks);
    passages[id] = { id, ambush: !!ambush, ambushTriggered: false, locked: false, blocked: false };
    a.exits.push({ to, passageId: id, connector, landmark,
      description: graphConnectorDescription(connector, landmark),
      keywords: [...graphConnectorVocabulary(connector), ...landmark.split(" ")] });
    const reverse = graphReverseConnector(connector);
    b.exits.push({ to: from, passageId: id, connector: reverse, landmark,
      description: graphConnectorDescription(reverse, landmark),
      keywords: [...graphConnectorVocabulary(reverse), ...landmark.split(" ")] });
  };

  const ids = areas.map(a => a.areaId).sort((a, b) => a - b);
  for (let i = 1; i < ids.length; i++) {
    const to = ids[i];
    // Prefer relatively nearby predecessors for an explorable branching layout.
    let parents = ids.slice(0, i).filter(id => nodes[id].exits.length < (id === ids[0] ? 3 : 4));
    const nearby = parents.filter(id => id >= to - 7);
    if (nearby.length) parents = nearby;
    const parent = randomEntry(parents);
    if (parent == null) throw new Error("Graph parent selection failed");
    const node = nodes[to];
    connect(parent, to, node.connector, node.ambush);
  }

  const requestedLoops = Math.floor(ids.length * 0.2);
  let addedLoops = 0;
  for (let attempt = 0; attempt < ids.length * 12 && addedLoops < requestedLoops; attempt++) {
    const a = randomEntry(ids);
    const b = randomEntry(ids);
    if (a === b || nodes[a].exits.length >= (a === ids[0] ? 3 : 4) || nodes[b].exits.length >= (b === ids[0] ? 3 : 4)) continue;
    if (nodes[a].exits.some(e => e.to === b)) continue;
    const connector = randomEntry(areaConnector);
    const ambush = connector.includes("20% chance of an ambush") && Math.random() < 0.2;
    connect(a, b, connector, ambush);
    addedLoops++;
  }

  const graph = {
    version: 1,
    nodes,
    passages,
    entryAreaId: ids[0],
    currentAreaId: null,
    mode: "AT_SHIP",
    navigationPending: null,
    encounterTrackingVersion: 1,
    entryPending: false,
    lastMove: null,
    showExits: false,
    totalAreas: ids.length
  };
  atdwEnsureShipNode(graph);
  atdwValidateSiteGraph(graph);
  log(`GRAPH BUILT | nodes=${ids.length} passages=${Object.keys(passages).length} loops=${addedLoops} entry=${ids[0]} ship=0 connected=true`);
  return graph;
};

/**
 * getSiteNode(site, areaId) / getSiteExits(site)
 * Return the authoritative node and its numbered, currently visible exits.
 * Exit numbers are local to the current room; they do not equal room IDs.
 */
const getSiteNode = (site, areaId) => site && site.nodes
  ? site.nodes[String(areaId)] || null : null;

const getSiteExits = (site) => {
  const node = getSiteNode(site, site && site.currentAreaId);
  if (!node) return [];
  return node.exits.map((exit, i) => ({ ...exit, number: i + 1,
    status: site.passages[exit.passageId] || {} }));
};

/**
 * formatGraphExitLine(site, exit)
 * Render a player-facing exit with TWO pieces of reliable orientation:
 *  1. A visited destination's actual Area number and persistent room title.
 *  2. The exact passage used to enter this area, when known.
 * Never reveal the identity of unexplored neighboring rooms. Passage IDs
 * survive navigation and are preferred over the legacy lastMove.from field.
 * This helper is shared by inline #EXITS output and optional entry popups.
 */
const formatGraphExitLine = (site, exit) => {
  const status = exit.status || {};
  const blocked = status.locked || status.blocked ? " [blocked]" : "";
  const destination = getSiteNode(site, exit.to);
  if (Number(exit.to) === 0) return `${exit.number}. ${exit.description} — EIRSEID SHIP [NODE 0]`;
  const title = destination && destination.visited ? getGraphRoomTitle(destination) : null;
  const knownDestination = destination && destination.visited
    ? ` — Area ${destination.areaId}${title ? ` — ${title}` : ""}`
    : "";
  const last = site && site.lastMove;
  const enteredThrough = !!(last && last.from != null &&
    Number(last.to) === Number(site.currentAreaId) &&
    (last.passageId != null
      ? String(last.passageId) === String(exit.passageId)
      : Number(last.from) === Number(exit.to)));
  return `${exit.number}. ${exit.description}${blocked}${knownDestination}` +
    (enteredThrough ? " [ENTERED FROM HERE]" : "");
};

/**
 * generateGraphNavContext(site)
 * Give CONTEXT exactly the current room plus short, non-spoiler descriptions
 * of neighboring areas. Full destination data is supplied separately as
 * hidden candidates for AI-assisted same-turn transitions.
 */
const generateGraphNavContext = (site) => {
  const node = getSiteNode(site, site.currentAreaId);
  if (!node) return "";
  return getSiteExits(site).map(exit => {
    const dest = getSiteNode(site, exit.to);
    const blocked = exit.status.blocked || exit.status.locked;
    return `${exit.number}. ${exit.description} ` +
      `(${dest.visited ? "explored" : "unexplored"}${blocked ? ", impassable" : ""})`;
  }).join("\n");
};

/**

/**
 * migrateGraphEncounterState(site)
 * One-time migration for saved Sites from the pre-V1 combat lifecycle.
 * In V1, a visit means that creature's introduction already happened;
 * neither an old "active" nor an old "resolved" label determines whether
 * it is currently alive. Future narration is the AI's responsibility.
 * Generated creature records and their cardCreated flags are NOT changed.
 */
const migrateGraphEncounterState = (site) => {
  if (!site || !site.nodes || site.encounterTrackingVersion === 1) return;
  Object.values(site.nodes).forEach(node => {
    if (node.encounterTriggered !== true && node.encounterTriggered !== false) {
      node.encounterTriggered = !!(node.encounterCreature &&
        (node.visited || node.encounterStatus === "active" ||
         node.encounterStatus === "resolved" || node.encounterResolved));
    }
    delete node.encounterStatus;
    delete node.encounterResolved;
  });
  delete site.encounterResolutionPending;
  site.encounterTrackingVersion = 1;
  log(`ENCOUNTER V1 MIGRATED | site=${site.siteId || "unknown"}`);
};

/**
 * triggerGraphEncounter(node, source)
 * Record exactly one creature introduction per area. No combat assessment,
 * victory flags, damage rolls, or AI metadata are required afterward.
 * Calling this function again never rerolls the creature or recreates a card.
 */
const triggerGraphEncounter = (node, source) => {
  if (!node || !node.encounterCreature || node.encounterTriggered) return false;
  node.encounterTriggered = true;
  log(`ENCOUNTER TRIGGERED | area=${node.areaId} name=${node.encounterCreature.creatureName} source=${source}`);
  return true;
};

// HP-01C5 — ONE REACTIVE SITE HAZARD (MECHANICAL TRAP)
// Only the exact pre-rolled mechanical trap from the established Site hazard
// table is mechanically supported. No regex over free AI narrative; no new
// hazard roll, surprise damage on discovery, or retroactive legacy triggering.
// A pending trap lets the player choose DEX evasion, Technology disarming,
// or leaving through an existing numbered passage without attempting a roll.
// ============================================================================
const EOE_MECHANICAL_TRAP_TEXT =
  "Trap: Mechanical trap: unstable flooring, crushing machinery, or entangling structures threaten an exposed character.";
const eoeSiteMechanicalTrapState = (site) => {
  if (!site || !site.nodes || site.mode!=="EXPLORING_ACTIVE") return null;
  const node=getSiteNode(site,site.currentAreaId);
  if (!node || node.result!=="Hazard Manifestation" ||
      node.hazardManifestation!==EOE_MECHANICAL_TRAP_TEXT) return null;
  const t=node.eoeMechanicalTrap;
  if (!t || t.schemaVersion!==1 || !["pending","resolved"].includes(t.status)) return null;
  return {site,node,trap:t};
};
// HP-01C6: location-scoped fixture for deterministic, player-invoked live tests.
// No generated node, encounter, hazard table, or health record is overwritten.
const eoeDebugTrapLocation = () => {
  const site=state.currentSiteId && state.sites && state.sites[state.currentSiteId];
  if (site && site.mode==="EXPLORING_ACTIVE") {
    const node=getSiteNode(site,site.currentAreaId);
    if (!node) return null;
    const key=`site:${String(site.siteId)}:area:${String(node.areaId)}`;
    return key.length<=160 ? {key,label:`Area ${node.areaId}`,siteId:site.siteId,areaId:node.areaId} : null;
  }
  if (site && (site.mode==="AT_SHIP" || state.travel && state.travel.phase==="arrived")) {
    const key=`site:${String(site.siteId)}:ship`;
    return key.length<=160 ? {key,label:"aboard the ship at this Site",siteId:site.siteId,areaId:null} : null;
  }
  const travel=state.travel || {};
  if (["traveling","event"].includes(travel.phase)) {
    const vessel=state.eoeCurrentShip && state.eoeCurrentShip.id || travel.shipId || "unnamed";
    const key=`voyage:${String(vessel)}:${String(travel.direction||"outbound")}`;
    return key.length<=160 ? {key,label:"aboard the ship during the current voyage",siteId:null,areaId:null} : null;
  }
  if (["idle","home"].includes(travel.phase))
    return {key:"hub:azaheim",label:"Azaheim",siteId:null,areaId:null};
  return null;
};
const eoeDebugTrapAtLocation = () => {
  const location=eoeDebugTrapLocation();
  const store=state.eoeDebugTraps;
  if (!location || !store || store.schemaVersion!==1 ||
      !store.locations || typeof store.locations!=="object") return null;
  const trap=Object.prototype.hasOwnProperty.call(store.locations,location.key)
    ? store.locations[location.key] : null;
  if (!trap || trap.schemaVersion!==1 ||
      !Number.isSafeInteger(trap.id) || trap.id<1 ||
      !["pending","resolved"].includes(trap.status)) return null;
  return {site:null,node:null,trap,source:"debug",location};
};
const eoeCurrentMechanicalTrap = site => {
  const debug=eoeDebugTrapAtLocation();
  const natural=eoeSiteMechanicalTrapState(site);
  // The current pending danger always has priority; a resolved fixture
  // never hides an unrelated naturally pending trap.
  if (debug && debug.trap.status==="pending") return debug;
  if (natural && natural.trap.status==="pending")
    return {...natural,source:"natural",location:{label:`Area ${natural.node.areaId}`}};
  if (debug) return debug;
  return natural ? {...natural,source:"natural",location:{label:`Area ${natural.node.areaId}`}} : null;
};
const eoeDebugTrapCommand = (operation,actionId) => {
  const error=reason=>({ok:false,error:reason});
  if (!Number.isSafeInteger(actionId) || actionId<0)
    return error("Debug trap commands require a stable action identity.");
  const location=eoeDebugTrapLocation();
  if (!location) return error("Current location is not supported for debug traps; no trap was created.");
  if (operation==="STATUS") {
    const current=eoeDebugTrapAtLocation();
    return {ok:true,message:current
      ? `DEBUG TRAP — ${location.label}: ${current.trap.status.toUpperCase()}. `+
        "Resolve with #TRAP EVADE (DEX) or #TRAP DISARM (Technology); remove with #DEBUG TRAP CLEAR."
      : `DEBUG TRAP — No test trap at ${location.label}. Use #DEBUG TRAP SPAWN to create one.`};
  }
  if (operation==="CLEAR") {
    const current=eoeDebugTrapAtLocation();
    if (!current) return error("No debug trap at the current location. Natural traps are never cleared by debug commands.");
    delete state.eoeDebugTraps.locations[location.key];
    return {ok:true,message:`DEBUG TRAP CLEARED — ${location.label}. No check, damage or changes to natural traps.`};
  }
  if (operation!=="SPAWN") return error("Use #DEBUG TRAP SPAWN, CLEAR, or STATUS.");
  const existing=eoeDebugTrapAtLocation();
  if (existing && existing.trap.status==="pending")
    return error("A test trap is already pending here. Resolve or clear it before spawning another.");
  const site=state.currentSiteId && state.sites && state.sites[state.currentSiteId];
  const natural=eoeSiteMechanicalTrapState(site);
  if (natural && natural.trap.status==="pending")
    return error("A natural trap is already pending here. Resolve it first; a debug trap will not conceal it.");
  let store=state.eoeDebugTraps;
  if (!store) store={schemaVersion:1,nextId:1,locations:{}};
  if (store.schemaVersion!==1 || !store.locations ||
      typeof store.locations!=="object" ||
      !Number.isSafeInteger(store.nextId) || store.nextId<1 ||
      store.nextId>=Number.MAX_SAFE_INTEGER)
    return error("Debug trap store invalid. Refusing to modify stored data.");
  const trap={schemaVersion:1,id:store.nextId,status:"pending",resolution:null};
  store.nextId++;
  store.locations[location.key]=trap;
  state.eoeDebugTraps=store;
  log(`DEBUG TRAP SPAWN | location=${location.key} id=${trap.id}`);
  return {ok:true,message:`DEBUG TRAP SPAWNED — ${location.label}. PENDING; no damage or roll. `+
    `Choose #TRAP EVADE (DEX), #TRAP DISARM (Technology), or #DEBUG TRAP CLEAR. `+
    `This is a test fixture; normal Site discoveries were not changed.`};
};

// HP-01C7 — LOCATION-SCOPED PROCEDURAL DEBUG CREATURE
// Reuse the canonical creature generator and health registry. Debug creatures
// receive randomized, difficulty-weighted Wound capacity once, while natural
// Site creatures remain unrated pending a later Mode-2 production bridge.
// No permanent Story Card: only current-location context contains this fixture,
// preventing old test creatures from consuming context after CLEAR.
// ============================================================================
// CBT-01D2 PROVISIONAL CREATURE TOUGHNESS. Six equally likely draws per
// selected combat difficulty. Each row has variance AND a rising mean:
// Easy: 1-3; Lightweight: 1-4; Average: 2-5; Hard: 2-6; Brutal: 3-7.
// This table is an Empire of Echoes adaptation, not an ATDW creature rule.
// Once assigned it NEVER changes with difficulty switches or Retry.
const EOE_CREATURE_WOUNDS_BY_DIFFICULTY = Object.freeze({
  1:Object.freeze([1,1,2,2,3,3]),
  2:Object.freeze([1,2,2,3,3,4]),
  3:Object.freeze([2,2,3,3,4,5]),
  4:Object.freeze([2,3,4,4,5,6]),
  5:Object.freeze([3,4,4,5,6,7])
});
const eoeRateDebugCreature = entry => {
  if (!entry || typeof entry.healthId!=="string" ||
      !/^creature:debug:[0-9]+$/.test(entry.healthId)) return null;
  const h=eoeHealthRecord(entry.healthId);
  if (!h || h.kind!=="CREATURE" || h.schemaVersion!==1 ||
      !Array.isArray(h.injuries) || !Array.isArray(h.conditions) ||
      !["alive","injured","dead"].includes(h.status)) return null;
  if (eoeHealthValidWoundRecord(h)) return h; // current rated Wounds are sacrosanct
  // Controlled migration ONLY for old alpha.12 unrated fixtures. Never
  // generate new health in place of corrupt, partially rated or dead data.
  if (h.maxWounds!==null || h.woundsRemaining!==null ||
      h.status!=="alive" || entry.status!=="present" ||
      h.woundEvents!==undefined || h.creatureRating!==undefined) return null;
  const level=state.atdwSettings && state.atdwSettings.combatDifficulty;
  const row=EOE_CREATURE_WOUNDS_BY_DIFFICULTY[level];
  if (!row) return null;
  const roll=1+Math.floor(Math.random()*6);
  const max=row[roll-1];
  h.maxWounds=max;h.woundsRemaining=max;
  h.creatureRating={schemaVersion:1,source:"CBT-01D2",difficulty:level,roll};
  log(`CREATURE TOUGHNESS | id=${entry.healthId} difficulty=${level} d6=${roll} wounds=${max}/${max}`);
  return h;
};

// CBT-01B — abstract two-position encounter state, independent of creature
// class, characteristics and equipment. No spatial grid or autonomous moves.
// Existing debug fixtures without this field default to ENGAGED so old saves
// preserve their previously usable immediate melee attacks.
const EOE_ENGAGEMENTS = ["AT_RANGE", "ENGAGED"];
const eoeCreatureEngagement = entry => entry && entry.engagement===undefined
  ? "ENGAGED" : entry && entry.engagement;
// Central modifier table for future PC rifle/melee attack integration.
// Shared engagement modifiers for PC and NPC attacks.
const eoeEngagementAttackModifier = (style,engagement) => {
  if (!EOE_ENGAGEMENTS.includes(engagement)) return null;
  if (style==="RIFLE") return engagement==="AT_RANGE"?2:-2;
  if (style==="MELEE") return engagement==="ENGAGED"?2:null;
  if (style==="RANGED") return engagement==="AT_RANGE"?2:-2;
  return null;
};
const eoeDebugRangeCommand = (operation, actionId) => {
  const error=reason=>({ok:false,error:reason});
  if (!Number.isSafeInteger(actionId) || actionId<0)
    return error("Range commands require a stable action identity; nothing changed.");
  const active=eoeDebugCreatureAtLocation();
  if (!active) return error("No valid debug creature at the current location; cannot change engagement.");
  const {entry,creature,location}=active;
  const health=eoeHealthRecord(entry.healthId);
  if (!health || health.kind!=="CREATURE" || health.status==="dead" ||
      entry.status!=="present" || health.schemaVersion!==1)
    return error("Creature is defeated or its identity is invalid; no engagement change.");
  if (operation!=="STATUS" && !EOE_ENGAGEMENTS.includes(operation))
    return error("Use #DEBUG RANGE STATUS, #DEBUG RANGE AT_RANGE, or #DEBUG RANGE ENGAGED.");
  // The fixture is debug-only. Setting distance is NOT a free production
  // disengage/close action; production rolls and autonomous pursuit come later.
  if (operation!=="STATUS") {
    entry.engagement=operation;
    // Explicit debug range placement also resets experimental escape state.
    if(entry.eoeEvasion) entry.eoeEvasion.phase='EXPOSED';
  }
  const pos=eoeCreatureEngagement(entry);
  const creatureStyle=creature.creatureClass==="Ranged"?"RANGED":"MELEE";
  const enemyModifier=eoeEngagementAttackModifier(creatureStyle,pos);
  const enemyText=creature.creatureClass==="Psychic"
    ? "Psychic attacks are not implemented"
    : enemyModifier===null
      ? "Melee creature cannot strike until it closes distance"
      : `${creatureStyle.toLowerCase()} creature attack modifier ${enemyModifier>=0?"+":""}${enemyModifier}`;
  const rifle=eoeEngagementAttackModifier("RIFLE",pos);
  return {ok:true,message:`DEBUG RANGE — ${creature.creatureName} at ${location.label}: ${pos}. `+
    `${enemyText}; hypothetical player rifle modifier ${rifle>=0?"+":""}${rifle} `+
    "(rifle reference only; prototype pistol/knife player attacks can resolve against test creatures). " +
    "Use #DEBUG RANGE AT_RANGE / ENGAGED to set test positioning; no roll, time, or damage."};
};
const eoeDebugCreatureCommand = (operation,actionId,requestedClass=null) => {
  const error=reason=>({ok:false,error:reason});
  if (!Number.isSafeInteger(actionId) || actionId<0)
    return error("Debug creature commands require a stable action identity.");
  const location=eoeDebugCreatureLocation();
  if (!location) return error("Current location is not supported for debug creatures; no creature was spawned.");
  const store=state.eoeDebugCreatures;
  if (store!==undefined && !eoeDebugCreatureStoreValid(store))
    return error("Debug creature store invalid. Refusing to modify stored data.");
  const existing=eoeDebugCreatureAtLocation();
  if (store && Object.prototype.hasOwnProperty.call(store.locations,location.key) && !existing)
    return error("Debug creature record invalid. Refusing to modify stored data.");
  if (operation==="STATUS") {
    if (!existing) return {ok:true,message:`DEBUG CREATURE — No test creature at ${location.label}. Use #DEBUG CREATURE SPAWN.`};
    const c=existing.creature;
    const h=eoeRateDebugCreature(existing.entry);
    if (!h) return error("Creature health record invalid; nothing changed.");
    const hp=`${h.woundsRemaining}/${h.maxWounds} Wounds remaining`;
    return {ok:true,message:`DEBUG CREATURE — ${c.creatureName} at ${location.label}. `+
      `Class: ${c.creatureClass}. Status: ${h.status.toUpperCase()}. ${hp}. `+
      `Difficulty at toughness roll: ${h.creatureRating?h.creatureRating.difficulty:"legacy"}. `+
      `Engagement: ${eoeCreatureEngagement(existing.entry)}. `+
      `Evasion: ${existing.entry.eoeEvasion ? existing.entry.eoeEvasion.phase : 'EXPOSED'}. `+
      `Trait: ${c.creatureTrait}. Ability: ${c.creatureAbility}. `+
      `Appearance: ${c.creaturePhysicalDescription} `+
      ((h.status==="dead")?"Dead: attack blocked. ":"Use #DEBUG CREATURE ATTACK for one opt-in strike. ")+
      `Director: ${(existing.entry.director&&existing.entry.director.enabled)?"ON":"OFF"}. `+
      "Use #DEBUG CREATURE AUTO ON/OFF/STATUS to test autonomy; #DEBUG CREATURE CLEAR to dismiss."};
  }
  if (operation==="CLEAR") {
    if (!existing) return error("No debug creature at the current location. Natural encounters are never removed.");
    delete store.locations[location.key];
    log(`DEBUG CREATURE CLEAR | location=${location.key} id=${existing.entry.id}`);
    return {ok:true,message:`DEBUG CREATURE CLEARED — ${existing.creature.creatureName} removed from test encounter at ${location.label}. `+
      "No combat, Wound, travel or natural encounter was changed."};
  }
  if (operation!=="SPAWN") return error("Use #DEBUG CREATURE SPAWN, STATUS, or CLEAR.");
  if (requestedClass!==null && !creatureClasses.some(c=>c.toUpperCase()===requestedClass))
    return error("Unknown creature class; use Brute, Lurker, Ranged, Swarm, or Psychic.");
  if (existing) return error(existing.entry.status==="defeated"
    ? "A defeated creature remains here. Use #DEBUG CREATURE CLEAR before spawning another."
    : "A debug creature is already present here. Use #DEBUG CREATURE CLEAR before spawning another.");
  if (!eoeCombatConfig()) return error("Combat difficulty invalid; no creature was generated.");
  if (!eoeHealthStore()) return error("Health registry unavailable; no creature was generated.");
  const next=store || {schemaVersion:1,nextId:1,locations:{}};
  // Check first, generate second: corrupt state must never consume a roll.
  const id=next.nextId, healthId=`creature:debug:${id}`;
  if (eoeHealthRecord(healthId)) return error("Creature identity already exists; refusing to overwrite it.");
  const creature=generateCreatureCard(false);
  if (requestedClass && creature) {
    creature.creatureClass=creatureClasses.find(c=>c.toUpperCase()===requestedClass);
    creature.creatureClassBehavior=creatureClassBehavior.find(b=>
      b.startsWith(`${requestedClass}:`)) ||
      `The creature behaves according to its ${creature.creatureClass} classification.`;
  }
  if (!creature || typeof creature.creatureName!=="string" || !creature.creatureName.trim())
    return error("Creature generation failed; no fixture was created.");
  const health=eoeHealthRegister(healthId,"CREATURE",creature.creatureName,null);
  if (!health) return error("Creature registration failed; no fixture was created.");
  const newEntry={schemaVersion:1,id,healthId,status:"present",engagement:"ENGAGED",creature};
  const rated=eoeRateDebugCreature(newEntry);
  if (!rated) return error("Creature toughness could not be assigned; fixture not activated.");
  next.locations[location.key]=newEntry;
  next.nextId++;
  state.eoeDebugCreatures=next;
  eoeRegisterCreatureSpecies(creature,true); // Stable species reference for debug specimens too.
  log(`DEBUG CREATURE SPAWN | location=${location.key} id=${id} class=${creature.creatureClass} wounds=${rated.maxWounds}`);
  return {ok:true,message:`DEBUG CREATURE SPAWNED — ${creature.creatureName} (${creature.creatureClass}) at ${location.label}. `+
    `${creature.creaturePhysicalDescription} Trait: ${creature.creatureTrait}. Ability: ${creature.creatureAbility}. `+
    `Wounds: ${rated.woundsRemaining}/${rated.maxWounds} (rolled once at difficulty ${rated.creatureRating.difficulty}). `+
    "Debug engagement starts ENGAGED (legacy behavior); use #DEBUG RANGE AT_RANGE to test separation. "+
    "The creature is now present for narrative encounters. No attack roll, combat Wound, or time advance occurred. " +
    "Use #DEBUG CREATURE STATUS to inspect it or #DEBUG CREATURE CLEAR to remove it. " +
    "Use #DEBUG CREATURE ATTACK for an opt-in strike; #DEBUG CREATURE AUTO ON enables test autonomy (OFF by default)."};
};
// ============================================================================
// CBT-01A — EXPLICIT DEBUG CREATURE ATTACK (not a finished ATDW combat engine)
// Uses an opposed d20: attacker provisional Attack rating 5, attacker +2;
// defender uses the PC's Close Combat for melee or DEX for ranged, including
// -2 per existing Wound exactly once. Psychic attacks are unmodeled, so
// they're rejected before rolling. This is an OPT-IN DEBUG DAMAGE BRIDGE:
// one hit means one Wound; final ATDW knockdown, weapon damage, armor,
// initiative, stamina, and creature numeric stats remain unimplemented.
// Only a present local debug fixture can attack. The action identity is fed
// into the existing authoritative Wound idempotency API.
// ============================================================================
const EOE_DEBUG_CREATURE_ATTACK_SKILL = 5; // provisional TEST rating, never a canonical creature stat
const eoeDebugCreatureAttack = actionId => {
  const error=reason=>({ok:false,error:reason});
  if (!Number.isSafeInteger(actionId) || actionId<0)
    return error("Creature attack requires a stable action ID; no rolls or injury.");
  const active=eoeDebugCreatureAtLocation();
  if (!active) return error("No debug creature is present at the current location; no attack.");
  const {entry,creature,location}=active;
  const health=eoeRateDebugCreature(entry);
  if (!health || health.kind!=="CREATURE" || health.schemaVersion!==1 ||
      health.status==="dead" || entry.status!=="present")
    return error(health && health.status==="dead"?"Creature is dead; no attack or roll.":"Creature combat record invalid; no attack.");
  if(entry.eoeEvasion && ["ESCAPED","HIDDEN"].includes(entry.eoeEvasion.phase))
    return error(`Creature cannot attack: encounter state ${entry.eoeEvasion.phase}.`);
  const profile=state.eoePlayerProfile;
  if (!profile || profile.kind!=="PC" || !profile.attributes || !profile.skills)
    return error("Player combat profile unavailable; no attack.");
  const eligibility=eoeHealthCanAct("pc");
  if (!eligibility.canAct) return error("Player health is unavailable or terminal; no attack.");
  const woundPenalty=eoeHealthWoundPenalty("pc");
  if (!Number.isSafeInteger(woundPenalty) || woundPenalty>0)
    return error("Player Wound penalty invalid; no attack.");
  const klass=creature.creatureClass;
  if (klass==="Psychic") return error("Psychic attacks require dedicated rules; use BRUTE, LURKER, RANGED, or SWARM for this test.");
  const ranged=klass==="Ranged";
  const engagement=eoeCreatureEngagement(entry);
  const style=ranged?"RANGED":"MELEE";
  const situation=eoeEngagementAttackModifier(style,engagement);
  if (situation===null)
    return error(`${creature.creatureName} is AT_RANGE and cannot make a melee attack; close distance first. No attack roll or injury.`);
  const defenseName=ranged ? "DEX" : "Close Combat";
  const defenseRating=ranged ? profile.attributes.DEX : profile.skills["Close Combat"];
  if (!Number.isSafeInteger(defenseRating) || defenseRating< -10 || defenseRating>30)
    return error(`Player ${defenseName} rating is unavailable; no attack.`);
  // Opposed check, not two independent DC20 tests. No creature rating was
  // synthesized from its trait/appearance or quietly saved to the registry.
  const difficulty=eoeCombatConfig();
  if (!difficulty) return error("Combat difficulty settings unavailable; no attack.");
  const attackerDie=Math.floor(Math.random()*20)+1;
  const defenderDie=Math.floor(Math.random()*20)+1;
  const creaturePenalty=eoeHealthWoundPenalty(entry.healthId);
  if(!Number.isSafeInteger(creaturePenalty)) return error("Creature Wound penalty invalid; no roll.");
  const attackerTotal=attackerDie+EOE_DEBUG_CREATURE_ATTACK_SKILL+2+situation+difficulty.enemyAttack+creaturePenalty;
  const defenderTotal=defenderDie+defenseRating+woundPenalty;
  let hit;
  if (attackerDie===20 && defenderDie!==20) hit=true;
  else if (attackerDie===1) hit=false;
  else if (defenderDie===20 && attackerDie!==20) hit=false;
  else if (attackerTotal!==defenderTotal) hit=attackerTotal>defenderTotal;
  else if (EOE_DEBUG_CREATURE_ATTACK_SKILL!==defenseRating) hit=EOE_DEBUG_CREATURE_ATTACK_SKILL>defenseRating;
  else {
    // Exact ties in this reduced prototype are inconclusive; do not invent
    // damage, and do not loop indefinitely if a debug RNG is fixed.
    hit=false;
  }
  let wound=null;
  if (hit) {
    wound=eoeHealthApplyWounds({id:"pc",wounds:1,
      cause:`Debug ${klass.toLowerCase()} creature attack by ${creature.creatureName}`,
      eventId:`cbt01a:creature:${entry.id}:action:${actionId}`});
    if (!wound || wound.ok!==true)
      return error("Health mutation refused the hit; no narrative Wound is claimed.");
  }
  const woundText=hit
    ? `HIT. Scripted debug rule: exactly 1 Wound applied; ${wound.remaining}/`+
      `${eoeHealthRecord("pc").maxWounds} remaining. ${wound.fatal?"PLAYER DIED.":"No knockdown modeled."}`
    : "DEFENDED / MISSED. No Wound inflicted.";
  return {ok:true,message:`DEBUG CREATURE ATTACK — ${creature.creatureName} (${klass}), ${location.label}\n`+
    `ATTACK: d20 ${attackerDie} + provisional Attack ${EOE_DEBUG_CREATURE_ATTACK_SKILL} + attacker bonus 2 `+
      `${situation>=0?"+":""}${situation} (${engagement.toLowerCase()} ${style.toLowerCase()})`+
      `${difficulty.enemyAttack ? ` ${difficulty.enemyAttack>=0?"+":""}${difficulty.enemyAttack} (difficulty ${state.atdwSettings.combatDifficulty} ${difficulty.name})` : ""}`+
      `${creaturePenalty ? ` ${creaturePenalty} (creature Wounds)` : ""} = ${attackerTotal}\n`+
    `ENGAGEMENT: ${engagement} — creature ${style.toLowerCase()} attack; prototype player attacks are separate.\n`+
    `DEFENSE: d20 ${defenderDie} + Player ${defenseName} ${defenseRating}`+
      `${woundPenalty ? ` ${woundPenalty} (Wounds)` : ""} = ${defenderTotal}\n`+
    `${woundText}\n`+
    "EXPERIMENTAL DEBUG ONLY: full ATDW damage, knocked-down, stamina, initiative and creature stats not implemented. " +
    "No autonomous debug reaction unless #DEBUG CREATURE AUTO ON was explicitly enabled."};
};

const eoeWeaponFind = text => {
  const t=String(text||"").toLowerCase();
  const found=[];
  for (const id of Object.keys(EOE_WEAPONS)) {
    const aliases=EOE_WEAPONS[id].aliases;
    if (aliases.some(a=>new RegExp('(?:^|[^a-z0-9])'+a+'(?=$|[^a-z0-9])','i').test(t))) found.push(id);
  }
  return [...new Set(found)];
};
const eoeWeaponEquip = name => {
  const inv=eoeWeaponInventory();
  if(!inv) return {ok:false,error:'Inventory state invalid; no changes made.'};
  const matches=eoeWeaponFind(name);
  if(matches.length!==1) return {ok:false,error:'Name exactly one recognized weapon; no equipment changed.'};
  const id=matches[0];
  if(!(inv.items[id]>0)) return {ok:false,error:`${EOE_WEAPONS[id].name} is not in the tracked inventory.`};
  inv.equipped=id;
  return {ok:true,message:`EQUIPPED — ${EOE_WEAPONS[id].name}. `+eoeWeaponInventoryDisplay()};
};
// Standalone, player-authored natural-language weapon selection. No rolls.
const eoeDetectNaturalEquip = raw => {
  const s=String(raw||'').trim().replace(/^>\s*(?:You\s+)?/i,'').trim();
  if(!s || s.length>180 || /[?#\n]/.test(s) || /^['"“‘]/.test(s) ||
     /\b(?:and|then|maybe|perhaps|consider|would|should|could|don't|do not|not|tell|ask)\b/i.test(s)) return null;
  if(!/^(?:i\s+)?(?:draw|ready|equip|unsheathe|take out|pull out)\s+(?:(?:my|the|a|an)\s+)?[\w -]+[.!]*$/i.test(s)) return null;
  const ids=eoeWeaponFind(s);
  return ids.length===1?ids[0]:null;
};
// Pure player-authored intent detector: abstains on descriptions, dialogue,
// third-party orders, speculative/negated actions, #commands, mixed attacks.
// Explicit verb may override normal weapon class, e.g. throw a knife.
const eoeDetectCombatAttempt = raw => {
  const source=String(raw||'').trim().replace(/^>\s*(?:You\s+)?/i,'').trim();
  if (!source || source.length>360 || /#[A-Za-z]/.test(source) ||
    /[?\n]/.test(source) || /^['"“‘]/.test(source)) return null;
  const s=source.toLowerCase().replace(/\s+/g,' ');
  if (/\b(?:do not|don't|never|won't|can't|cannot|wouldn't|not going to|rather than|instead of|without|stop myself from)\b/.test(s) ||
      /\b(?:maybe|perhaps|might|could|should|would|consider|thinking of|plan to|intend to|want to|wish|later|if i|what if|pretend|imagine)\b/.test(s) ||
      /^(?:i|we)\s+(?:ask|tell|order|instruct|request|command|suggest|quote|describe|report|think|wonder|discuss|watch|read|remember)\b/.test(s) ||
      /^(?:bianca|sam|he|she|they|someone|anyone|the crew|you)\b/.test(s) ||
      /^(?:i|we)\s+(?:say|shout|yell|whisper)\b/.test(s)) return null;
  // Require player-initiated immediate action rather than incidental verbs.
  const actor=s.match(/^(?:i\s+|we\s+|my\s+)?(?:try\s+to\s+|attempt\s+to\s+|quickly\s+|carefully\s+)?(.*)$/);
  const action=actor && actor[1] || s;
  if (/^(?:aim|ready|draw|holster|reload|lower|raise|hold|inspect|point|equip|prepare)\b/.test(action) &&
      !/\b(?:and|then)\s+(?:shoot|fire|throw|hurl|swing|slash|stab|strike|hit|bash|smash|attack)\b/.test(action)) return null;
  // Permit preparatory draw AND shoot; do not trigger combat merely on draw.
  const shot=/\b(?:fire|shoot|blast)\b/.test(action);
  const toss=/\b(?:throw|hurl|toss)\b/.test(action);
  const melee=/\b(?:hit|strike|swing|slash|stab|thrust|bash|smash|punch)\b/.test(action);
  if (Number(shot)+Number(toss)+Number(melee)!==1) return null;
  const intent=shot?'RANGED':toss?'THROWN':'MELEE';
  // Targeting must attach to the attack verb/preposition. Mere mention of
  // a creature elsewhere (e.g. shooting the wall while observing it) is not
  // permission to attack it. Warning shots are not creature attacks.
  const noun='(?:creature|beast|monster|thing|alien|enemy|brute|lurker|swarm|predator|it|them)';
  const activeTarget=eoeDebugCreatureAtLocation();
  const targetName=activeTarget
    ? activeTarget.creature.creatureName.toLowerCase().trim() : "";
  const escapedName=targetName.replace(/[.*+?^${}()|[\]\\]/g,'\\$&');
  const directed=new RegExp('\\b(?:at|toward|towards|into|against)\\s+(?:(?:the|a|an|that)\\s+)?'+noun+'\\b').test(action) ||
    new RegExp('\\b(?:shoot|stab|strike|hit|slash|smash|punch|thrust|bash)\\s+(?:(?:the|a|an|that)\\s+)?'+noun+'\\b').test(action) ||
    (escapedName && new RegExp('\\b(?:at|toward|towards|into|against|shoot|stab|strike|hit|slash|smash|punch|thrust|bash)\\s+(?:(?:the|a|an|that)\\s+)?'+escapedName+'\\b').test(action));
  if(!directed || /\b(?:warning shot|shot into the air|fire into the air)\b/.test(action)) return null;
  const items=eoeWeaponFind(action);
  if(items.length>1) return null;
  const inv=eoeWeaponInventory();
  if(!inv) return {ok:false,error:'Inventory state invalid; no combat roll.'};
  const weaponId=items.length?items[0]:inv.equipped;
  if(!weaponId) return null;
  if(!(inv.items[weaponId]>0)) return {ok:false,error:`${EOE_WEAPONS[weaponId].name} is not in your tracked inventory.`};
  const w=EOE_WEAPONS[weaponId];
  if (intent==='THROWN' && weaponId!=='combat_knife') return null;
  if (intent==='RANGED' && w.style!=='RANGED') return null;
  const style=intent==='RANGED' || intent==='THROWN'?'RANGED':'MELEE';
  // A melee strike with a firearm is allowed as an improvised close attack.
  return {ok:true,weaponId,style,skill:style==='MELEE'?'Close Combat':'Ranged Combat',
    thrown:intent==='THROWN',attempt:source.slice(0,240)};
};
const eoeResolveCombatAttempt = (detected,actionId) => {
  const error=reason=>({ok:false,error:reason,actionId});
  if (!Number.isSafeInteger(actionId) || actionId<0)
    return error('An action identity is required to roll safely.');
  if (!detected || !detected.ok) return error(detected && detected.error || 'Unrecognized attack.');
  const target=eoeDebugCreatureAtLocation();
  if(!target) return error('No supported test creature is present here. No attack rolled.');
  if (target.entry.status==="defeated") return error('Creature is already dead; no attack rolled.');
  if(target.entry.eoeEvasion && target.entry.eoeEvasion.phase==='ESCAPED') return error('Encounter escaped; no target in immediate combat.');
  const health=eoeRateDebugCreature(target.entry);
  if (!eoeHealthValidWoundRecord(health) || health.kind!=='CREATURE' || health.status==='dead')
    return error('Creature combat record invalid; no roll.');
  const inv=eoeWeaponInventory();
  if (!inv || !(inv.items[detected.weaponId]>0)) return error('Weapon no longer available.');
  const profile=state.eoePlayerProfile;
  if (!profile || profile.kind!=='PC' || !profile.skills || !eoeHealthCanAct('pc').canAct)
    return error('Player combat profile unavailable.');
  const score=profile.skills[detected.skill];
  const penalty=eoeHealthWoundPenalty('pc');
  if(!Number.isSafeInteger(score) || !Number.isSafeInteger(penalty) || penalty>0)
    return error(`${detected.skill} skill or Wound penalty unavailable; no roll.`);
  const engagement=eoeCreatureEngagement(target.entry);
  // CBT-01D5: the FIRST genuine attack from concealment earns one +2
  // situational bonus. Revealing happens on this resolved attempt, HIT or MISS;
  // failed/rejected attempts cannot consume concealment.
  const hiddenAtAttack=!!(target.entry.eoeEvasion && target.entry.eoeEvasion.phase==='HIDDEN');
  const hiddenAttackBonus=hiddenAtAttack?2:0;
  const situation=eoeEngagementAttackModifier(detected.style,engagement);
  if(situation===null) return error('The creature is AT_RANGE; a melee strike cannot reach it. No roll.');
  // This opponent defense is explicitly provisional, NOT official ATDW data.
  const defense=5;
  const attackDie=Math.floor(Math.random()*20)+1;
  const defenseDie=Math.floor(Math.random()*20)+1;
  const attackTotal=attackDie+score+2+situation+penalty+hiddenAttackBonus;
  const defenseTotal=defenseDie+defense;
  let hit=false;
  if(attackDie===20 && defenseDie!==20)hit=true;
  else if(attackDie===1)hit=false;
  else if(defenseDie===20 && attackDie!==20)hit=false;
  else if(attackTotal!==defenseTotal)hit=attackTotal>defenseTotal;
  else hit=score>defense;
  let wound=null;
  if(hit){
    wound=eoeHealthApplyWounds({id:target.entry.healthId,wounds:1,
      cause:`${detected.weaponId} hit by player`,
      eventId:`cbt01d2:player:${actionId}:target:${target.entry.healthId}`});
    if(!wound || !wound.ok || wound.duplicate)
      return error('Creature Wound could not be recorded safely; no equipment changed.');
    if(wound.fatal) target.entry.status="defeated";
  }
  if(hiddenAtAttack) target.entry.eoeEvasion.phase='EXPOSED';
  if (!detected.thrown) inv.equipped=detected.weaponId; // player used it explicitly
  // Thrown knives leave the inventory regardless of hit. No automatic
  // recovery is claimed; retrieval is deferred to later item interaction.
  if(detected.thrown){
    inv.items.combat_knife--;
    if(inv.equipped==='combat_knife') inv.equipped=null; // never auto-draw a pistol
  }
  return {ok:true,actionId,weaponId:detected.weaponId,weaponName:EOE_WEAPONS[detected.weaponId].name,
    style:detected.style,skill:detected.skill,attempt:detected.attempt,thrown:detected.thrown,
    targetName:target.creature.creatureName,targetClass:target.creature.creatureClass,
    targetDrive:target.creature.creatureDrive,targetBehavior:target.creature.creatureClassBehavior,
    engagement,attackDie,defenseDie,score,penalty,situation,hiddenAtAttack,hiddenAttackBonus,attackTotal,defenseTotal,hit,
    creatureWoundApplied:wound ? wound.applied:0,
    creatureWoundsRemaining:health.woundsRemaining,
    creatureMaxWounds:health.maxWounds,
    creatureDead:health.status==="dead"};
};

const eoeDetectEvasionIntent = raw => {
  const src=String(raw||'').trim().replace(/^>\s*(?:You\s+)?/i,'').trim();
  if(!src || src.length>300 || /[?#\n]/.test(src) || /^['"“‘]/.test(src)) return null;
  const v=src.toLowerCase().replace(/\s+/g,' ');
  if(/\b(?:don't|do not|not|never|without|would|should|could|might|maybe|perhaps|if i|consider|thinking|want to|plan to|tell|ask|order|watch|observe|pretend|imagine)\b/.test(v) ||
     /^(?:bianca|sam|they|she|he|the creature|you)\b/.test(v) ||
     /\b(?:fire|shoot|stab|strike|swing|slash|attack|throw|hurl)\b/.test(v)) return null;
  const a=v.replace(/^(?:i\s+|we\s+)?(?:try\s+to\s+|attempt\s+to\s+|quickly\s+)?/,'');
  // Recognize intent even when it follows a preparatory clause: 'I remain
  // behind the crate and try to conceal myself' is a full HIDE attempt.
  const hide=/\b(?:hide|hiding|conceal(?:ed|ing)?(?:\s+myself)?|sneak|slip\s+(?:into|behind|under)|take\s+cover|hunker\s+down|keep\s+out\s+of\s+sight|stay\s+(?:hidden|concealed)|remain\s+(?:hidden|concealed)|blend\s+into\s+(?:the\s+)?shadows)\b/.test(a) ||
    /^(?:duck|crouch)\b/.test(a) && /\b(?:behind|under|cover|shadow|crate|machinery|conceal)\b/.test(a);
  const flee=/\b(?:flee|sprint|dash|bolt|retreat|escape|disengage|run\s+away|make\s+a\s+break|pull\s+back|back\s+away)\b/.test(a) ||
    /^(?:run|race|rush)\b/.test(a) && /\b(?:away|exit|door|corridor|hall|ship|safety|out|past|distance)\b/.test(a);
  if(hide===flee) return null;
  const cover=/\b(?:behind|under|beneath|inside|within|cover|machinery|crate|container|debris|shadow|darkness|wall|corner|alcove|doorway|locker|cabinet|duct|vent|shelter)\b/.test(a);
  return {kind:hide?'HIDE':'FLEE',cover,attempt:src.slice(0,300)};
};

const eoeResolveEvasion = (intent,actionId) => {
  const fail=error=>({ok:false,error,actionId,kind:intent&&intent.kind});

  if(!d) return fail('Director record invalid; no check rolled.');
  if(active.entry.eoeEvasion && active.entry.eoeEvasion.phase==='ESCAPED')
    return fail('Immediate encounter already escaped.');
  if(intent.kind==='HIDE' && !intent.cover)
    return fail('No explicit concealment or cover in the attempted action; no hide roll.');

  const faster=/\bfaster\b/i.test(c.creatureTrait||'')?2:0;

  const situational=intent.kind==='HIDE'&&engagement==='ENGAGED'?-2:0;

  const total=die+score+penalty+situational,opposedTotal=enemyDie+opposedScore;

    enemyDie===20?false:total>opposedTotal;

  const result={ok:true,actionId,kind:intent.kind,attempt:intent.attempt,skill,
    die,enemyDie,score,penalty,situational,opposedScore,total,opposedTotal,
    success,phase,engagementBefore:before,engagementAfter:after,
    creatureName:c.creatureName,drive};
  log(`CBT EVASION | action=${actionId} kind=${intent.kind} success=${success} `+
      `roll=${die}+${score}${penalty<0?penalty:''}${situational<0?situational:''}=${total} `+
      `vs=${enemyDie}+${opposedScore}=${opposedTotal} phase=${phase} range=${before}->${after}`);
  return result;
};

const eoeReconcileEvasionProse = (prose,evasion,director) => {
  const s=String(prose||'').trim();
  if(!evasion || !evasion.ok || evasion.success || !s) return s;
  const fleeing=evasion.kind==='FLEE';
  const conflicting=fleeing
    ? /\b(?:you (?:(?:reach|reached|make it to|made it to|arrive at|arrived at)\s+(?:the\s+|a\s+|an\s+)?(?:corridor|junction|exit|doorway|door|ship|stairs|hall|bay|safe area)|escape(?:d)?(?: from)?|get clear|got clear|gain(?:ed)? distance|put distance between|break free|broke free)|you(?:'ve| have) (?:escaped|gotten clear)|the (?:creature|beast|monster|\w+) (?:stops|stopped|remains|waits|watches) (?:at|behind|by) the (?:mouth|end|entrance|exit)|you (?:are|were) (?:safely )?(?:out of reach|clear of the creature))\b/i.test(s)
    : /\b(?:the creature (?:loses|lost) (?:sight|track) of you|it (?:cannot|can't|doesn't) see you|you (?:are|remain|stay) hidden|you (?:have )?(?:vanished|disappeared) from (?:its|the creature's) (?:sight|view))\b/i.test(s);
  if(!conflicting) return s;
  // Contradictory outcome prose is replaced, not patched with another
  // contradictory sentence. These are consequence descriptions, not newly
  // invented player decisions or relocated player/creature positions.
  let scene=fleeing
    ? 'Your attempt to get away is cut short. The creature keeps pace and remains close enough to strike; the exit is still beyond your reach.'
    : 'You try to disappear into cover, but the creature keeps you in sight. You remain exposed.';
  if(director && director.kind==='STRIKE') {
    scene+=director.hit
      ? ' The creature attacks and lands a blow, leaving you wounded.'
      : ' The creature lashes out, but its attack misses.';
  } else if (director && director.kind==='WARN') scene+=' The creature answers with a hostile warning.';
  return scene;
};

  if(ev && ev.phase==='ESCAPED') return null;
  if(ev && ev.phase==='HIDDEN') {
    const findRoll=Math.floor(Math.random()*20)+1;
    const found=findRoll>=17; // provisional searching rule, no surprise strike
    if(found) {ev.phase='EXPOSED';d.posture='WARN';}
    else d.posture='SEARCH';
    d.beats++;
    const outcome={schemaVersion:1,actionId,kind:found?'WARN':'SEARCH',cause,
      creatureName:creature.creatureName,creatureClass:creature.creatureClass,
      creatureDrive:creature.creatureDrive,creatureTrait:creature.creatureTrait,
      creatureAbility:creature.creatureAbility,engagementBefore:eoeCreatureEngagement(entry),
      engagementAfter:eoeCreatureEngagement(entry),creatureWounds:hp.woundsRemaining,
      creatureMaximum:hp.maxWounds,searchFound:found,provisional:true};
    d.lastActionId=actionId;d.lastEvent=outcome;state.eoeCreatureDirectorTurn=outcome;
    log(`CBT DIRECTOR | action=${actionId} choice=${outcome.kind} searchFound=${found} no-attack`);
    return outcome;
  }
  const klass=creature.creatureClass;
  const drive=String(creature.creatureDrive||'').split(':')[0].trim().toUpperCase();
  const engagement=eoeCreatureEngagement(entry);
  const playerAttacked=cause==='PLAYER_ATTACK';
  if (playerAttacked) d.provoked=true;
  const rand=Math.random();
  let kind='HOLD';
  const injured=hp.woundsRemaining<hp.maxWounds;
  const seriouslyInjured=hp.woundsRemaining<=Math.ceil(hp.maxWounds/2);
  // Personality always matters. Territorial creatures warn/hold ground;
  // lurkers stalk, ranged attackers seek distance, predators close for prey,
  // destructive creatures escalate, parasitic creatures look for a host.
  if (klass==='Psychic') kind='PSYCHIC_MENACE';
  else if (d.cooldown>0) {kind=klass==='Lurker' && engagement==='ENGAGED'?'WITHDRAW':
    drive==='TERRITORY CONTROL'?'WARN':'HOLD';d.cooldown--;}
  else if (seriouslyInjured && drive!=='DESTRUCTION' && rand<.30) kind='WITHDRAW';
  else if (klass==='Ranged' && engagement==='ENGAGED') kind='WITHDRAW';
  else if (klass!=='Ranged' && engagement==='AT_RANGE') {
    if (klass==='Lurker' && !d.provoked && d.beats<2) kind='STALK';
    else if (drive==='TERRITORY CONTROL' && !d.provoked && d.beats<2) kind='WARN';
    else kind=rand<.78?'APPROACH':(klass==='Lurker'?'STALK':'WARN');
  }
  else if (drive==='TERRITORY CONTROL' && !d.provoked && d.beats<2) kind='WARN';
  else if (klass==='Lurker' && !d.provoked && rand<.55) kind='STALK';
  else if (drive==='PARASITIZATION' && !d.provoked && rand<.52) kind='STALK';
  else if (drive==='PREDATION' && !d.provoked && d.beats===0) kind='STALK';
  else if (drive==='DESTRUCTION') kind=rand<.78?'STRIKE':'WARN';
  else if (d.provoked) kind=rand<.70?'STRIKE':'WARN';
  else kind=rand<.48?'STRIKE':'WARN';
  // Only actual physical strikes use the modeled opposed roll. A menacing
  // ability/venom/psychic display never applies arbitrary damage or conditions.
  let strike=null;
  if (kind==='STRIKE') {
    strike=eoeCreatureDirectorStrike(active,actionId);
    if (!strike) kind='HOLD';
    else d.cooldown=1;
  }
  if (kind==='APPROACH') entry.engagement='ENGAGED';
  if (kind==='WITHDRAW') entry.engagement='AT_RANGE';
  d.beats++;
  d.posture=kind;
  const event={schemaVersion:1,actionId,kind,cause,creatureName:creature.creatureName,
    creatureClass:klass,creatureDrive:drive,creatureTrait:creature.creatureTrait,
    creatureAbility:creature.creatureAbility,engagementBefore:engagement,
    engagementAfter:eoeCreatureEngagement(entry),creatureWounds:hp.woundsRemaining,
    creatureMaximum:hp.maxWounds,hit:strike?strike.hit:undefined,
    strike,provisional:true};
  d.lastActionId=actionId;d.lastEvent=event;
  state.eoeCreatureDirectorTurn=event;
  log(`CBT DIRECTOR | action=${actionId} name=${creature.creatureName} drive=${drive} class=${klass} `+
    `choice=${kind} range=${event.engagementBefore}->${event.engagementAfter}`+
    (strike?` attack=${strike.hit?'HIT':'MISS'} wound=${strike.woundApplied} pc=${strike.playerRemaining}/${strike.playerMaximum}`:''));
  return event;
};
const eoeCreatureDirectorContext = event => {
  if (!event) return '';
  const e=event,who=e.creatureName;
  const actions={WARN:'warns, displays hostility or guards its area without attacking',
    STALK:'stalks, watches or seeks an opening without attacking',
    SEARCH:'searches for the hidden player; no attack or guaranteed discovery',
    APPROACH:'closes the abstract distance to ENGAGED; no attack this beat',
    WITHDRAW:'moves back to AT_RANGE; no attack this beat',
    HOLD:'hesitates, watches or repositions narratively; no attack',
    PSYCHIC_MENACE:'unsettling psychic presence is atmospheric ONLY; no psychic attack or condition',
    STRIKE:'launches ONE resolved physical attack against the player'};
  return `AUTHORITATIVE CREATURE DIRECTOR: ${who}, class ${e.creatureClass}, `+
    `drive ${e.creatureDrive}, trait ${e.creatureTrait}, ability ${e.creatureAbility}. `+
    `Decision ${e.kind}: ${actions[e.kind]||'waits'}. `+
    `Range ${e.engagementBefore} -> ${e.engagementAfter}. `+
    (e.strike?`Opposed roll: creature ${e.strike.die} -> ${e.strike.total}, player `+
      `${e.strike.defenseDie} -> ${e.strike.defenseTotal}; ${e.strike.hit?'CREATURE HIT':'CREATURE MISSED'}; `+
      `${e.strike.woundApplied} actual player Wounds, `+
      `${e.strike.playerRemaining}/${e.strike.playerMaximum} remaining. `:
      'NO attack roll, player Wound or creature special-effect damage this beat. ')+
    `Narrate this one creature action naturally, respecting its Drive and Class. `+
    `Do not invent a second attack, player action, retreat, wounds, poison, infection, weapon effects or different range. `+
    `The player controls every voluntary action; environmental effects are allowed only if supported by scripted results. `+
    `No numeric roll exposition in ordinary prose.`;
};
const eoeCreatureDirectorOutput = (prose,event) => {
  if (!event) return prose;
  const s=String(prose||'').trim();
  if (s) {
    // A missed or silent narrator acknowledgement cannot erase an actual PC Wound.
    // Avoid appending redundant mechanical text when prose already describes it.
    if (event.strike && event.strike.hit &&
        !/\b(?:you(?:'re| are)? (?:wound|injur|hurt|hit|struck|bleed)|your (?:arm|leg|shoulder|side|chest|body|blood)|pain (?:shoots?|flares?|rips?|explodes?)|strikes? you|catches you|hits? you|claws? (?:you|your)|tears? into you|slams? into you)\b/i.test(s))
      return `${s}\nThe ${String(event.creatureName||'creature').toLowerCase()} attacks, leaving you wounded.`;
    return s;
  }
  const who='The '+String(event.creatureName||'creature').toLowerCase();
  if (event.kind==='STRIKE') return event.hit
    ? `${who} strikes you. The blow causes one Wound; ${event.strike.playerRemaining}/${event.strike.playerMaximum} remain.`
    : `${who} lunges at you, but the attack fails to connect.`;
  const cues={WARN:`${who} gives a threatening warning but holds position.`,
    STALK:`${who} moves cautiously, searching for an opening.`,
    SEARCH:`${who} searches its surroundings, unable to pinpoint the hidden player.`,
    APPROACH:`${who} advances and closes the distance.`,
    WITHDRAW:`${who} withdraws out of immediate melee reach.`,
    HOLD:`${who} watches, tense and uncertain.`,
    PSYCHIC_MENACE:`${who} projects an unsettling presence without striking.`};
  return cues[event.kind]||`${who} hesitates.`;
};

// CBT-01D4 site lifecycle: one departure snapshot; no offscreen simulation
// after departure. A Site's known inhabitants are possibilities, not an active
// immortal monster or a promise that the same individual will return.
const eoeArchiveSiteInhabitants = site => {
  if(!site || !site.siteId || !site.nodes || site.mode!=='AT_SHIP') return false;
  if(site.eoeInhabitantArchive && site.eoeInhabitantArchive.schemaVersion!==1) return false;
  const names=new Set(site.eoeInhabitantArchive && Array.isArray(site.eoeInhabitantArchive.species)?
    site.eoeInhabitantArchive.species:[]);
  // Only encounters already discovered by the player; no hidden-site spoilers.
  Object.keys(site.nodes).forEach(id=>{
    const n=site.nodes[id];
    if(n && n.encounterTriggered && n.encounterCreature &&
       typeof n.encounterCreature.creatureName==='string') names.add(n.encounterCreature.creatureName);
  });
  const prefix=`site:${String(site.siteId)}:`;
  const store=state.eoeDebugCreatures;
  if(store && eoeDebugCreatureStoreValid(store)) {
    Object.keys(store.locations).forEach(key=>{
      if(!key.startsWith(prefix)) return;
      const entry=store.locations[key];
      const h=entry&&entry.healthId?eoeHealthRecord(entry.healthId):null;
      if(entry && entry.creature && entry.status==='present' && h && h.status!=='dead')
        names.add(entry.creature.creatureName);
      delete store.locations[key]; // retire fixtures; NEVER run them offsite
    });
  }
  const species=[...names].filter(x=>typeof x==='string'&&x.trim()).slice(0,32);
  site.eoeInhabitantArchive={schemaVersion:1,species,simulationActive:false};
  if(typeof storyCards==='undefined'||!Array.isArray(storyCards)||!species.length) return true;
  // Preserve existing custom card content and append only one idempotent note.
  const idx=storyCards.findIndex(c=>c && typeof c.keys==='string' &&
    c.keys.startsWith(`${site.name}, site, exploration, location, explore`));
  if(idx<0 || typeof updateStoryCard!=='function') return true;
  const card=storyCards[idx],mark='[EOE POSSIBLE INHABITANTS]';
  const current=String(card.entry||'');
  const base=current.includes(mark)?current.slice(0,current.indexOf(mark)).trimEnd():current.trimEnd();
  const note=`${mark}\nSpecies encountered here (may inhabit the Site on a later visit, not guaranteed): ${species.join(', ')}. `+
    `No active creature simulation while the crew is away. `+
    `Encountered creatures must not be assumed alive or revived if killed.\n`;
  updateStoryCard(idx,card.keys,base+'\n\n'+note,card.type,card.name);
  log(`SITE INHABITANTS ARCHIVED | site=${site.siteId} species=${species.length} sim=OFF`);
  return true;
};

const eoeArmSiteMechanicalTrap = (site,node,firstVisit) => {
  if (!firstVisit || !site || !node || node.result!=="Hazard Manifestation" ||
      node.hazardManifestation!==EOE_MECHANICAL_TRAP_TEXT ||
      Object.prototype.hasOwnProperty.call(node,"eoeMechanicalTrap")) return false;
  node.eoeMechanicalTrap={schemaVersion:1,status:"pending",resolution:null};
  log(`SITE TRAP ARMED | site=${site.siteId} area=${node.areaId}`);
  return true;
};
const eoeResolveSiteMechanicalTrap = (site,choice,actionId) => {
  const error=reason=>({ok:false,error:reason});
  if (!Number.isSafeInteger(actionId) || actionId<0)
    return error("Trap resolution requires a stable action identity.");
  if (!["EVADE","DISARM"].includes(choice))
    return error("Use #TRAP EVADE (DEX) or #TRAP DISARM (Technology).");
  const current=eoeCurrentMechanicalTrap(site);
  if (!current) return error("No scripted mechanical trap is active at this location.");
  const {node,trap,source,location}=current;
  if (trap.status!=="pending") return error("This trap has already been resolved; it cannot trigger again.");
  if (!state.eoePlayerProfile || state.eoePlayerProfile.kind!=="PC")
    return error("Player skill profile unavailable; no roll or injury.");
  if (!eoeHealthCanAct("pc").canAct) return error("The player cannot act; no trap roll.");
  const penalty=eoeHealthWoundPenalty("pc");
  if (penalty===null) return error("Player Wound state invalid; no roll or injury.");
  const record=eoeHealthRecord("pc");
  if (!eoeHealthValidWoundRecord(record)) return error("Invalid health record; no roll or injury.");
  const checkName=choice==="EVADE"?"DEX":"Technology";
  // Prevalidate score to avoid a die roll or half-committed resolution.
  const actor=state.eoePlayerProfile;
  const score=checkName==="DEX" ? actor.attributes && actor.attributes.DEX :
    actor.skills && actor.skills.Technology;
  if (!Number.isSafeInteger(score)) return error(`${checkName} score unavailable; no roll or injury.`);
  const eventId=source==="debug" ? `debug-trap:${trap.id}` :
    `trap:${String(site.siteId)}:area:${String(node.areaId)}`;
  if (eventId.length>160) return error("Trap identity invalid; no roll or injury.");
  const check=eoeResolveCheck({actor,check:checkName,difficulty:"normal",woundPenalty:penalty});
  if (!check.ok) return error(`Trap check not rolled: ${check.error}`);
  let wound=null;
  if (!check.success) {
    wound=eoeHealthApplyWounds({id:"pc",wounds:1,eventId,
      cause:source==="debug" ? `debug test mechanical trap at ${location.label}` :
        "mechanical trap in Area "+node.areaId});
    // Do not label the trap resolved if an unsupported/corrupt Wound state
    // could not record the failure's damage. Normal safeguards run pre-roll.
    if (!wound.ok) return error(`Trap result could not be committed safely: ${wound.error}`);
  }
  const result={ok:true,actionId,siteId:source==="debug"?location.siteId:site.siteId,
    areaId:source==="debug"?location.areaId:node.areaId,locationLabel:location.label,
    source,choice,checkName,check,success:check.success,
    woundApplied:wound ? wound.applied : 0,
    remaining:wound ? wound.remaining : record.woundsRemaining,
    maxWounds:record.maxWounds,
    status:check.success?(choice==="DISARM"?"disarmed":"avoided"):"sprung"};
  trap.status="resolved";
  trap.resolution=result;
  log(`SITE TRAP RESOLVED | source=${source} location=${location.label} choice=${choice} die=${check.die} success=${check.success} wound=${result.woundApplied}`);
  return result;
};
const eoeSiteTrapReport = turn => {
  if (!turn || !turn.ok)
    return `SITE TRAP — NO ACTION\n${turn && turn.error || "Unknown trap command."}`;
  if (turn.query) return turn.message;
  const checkTurn={ok:true,actorName:"Player",check:turn.checkName,result:turn.check};
  return `${turn.source==="debug"?"DEBUG TEST TRAP":"SITE TRAP"} — ${turn.locationLabel||`Area ${turn.areaId}`}\n`+
    `Action: ${turn.choice=== "EVADE" ? "Evade the mechanism" : "Disarm the mechanism"}\n`+
    eoeSkillCheckResultText(checkTurn)+"\n"+
    (turn.success
      ? (turn.choice==="DISARM" ? "The mechanism is disarmed. No Wound taken." : "You avoid the trap. No Wound taken.")
      : `The mechanism springs. Player takes exactly ${turn.woundApplied} Wound; ${turn.remaining}/${turn.maxWounds} remaining. `+
        `You may use #TREAT Player for a separate Medical Aid attempt if alive.`)+
    "\nThis trap is resolved and will not strike again.";
};

/**
 * commitGraphMovement(site, exitNumber, source)
 * Validate an exit against the current node and shared passage conditions.
 * Mutates ONLY the Site's current position and visit flags; it does not
 * resolve encounters, hazards, or artifacts. Re-visits never re-roll data.
 * Returns false for an invalid or obstructed exit.
 */
const commitGraphMovement = (site, exitNumber, source) => {
  const from = getSiteNode(site, site.currentAreaId);
  if (!from) { log("NAV BLOCK | no current node"); return false; }
  const exit = getSiteExits(site).find(e => e.number === Number(exitNumber));
  if (!exit) { log(`NAV BLOCK | exit=${exitNumber} does not exist`); return false; }
  if (exit.status.locked || exit.status.blocked) {
    log(`NAV BLOCK | exit=${exitNumber} locked=${!!exit.status.locked} blocked=${!!exit.status.blocked}`);
    return false;
  }
  const dest = getSiteNode(site, exit.to);
  if (!dest) { log(`NAV BLOCK | missing destination=${exit.to}`); return false; }
  if (dest.areaId === 0) {
    site.mode = "AT_SHIP";
    site.currentAreaId = 0;
    site.lastMove = { from: from.areaId, to: 0, exitNumber: exit.number,
      passageId: exit.passageId, connector: exit.connector, source, needsNarration: true,
      firstVisit: false, ambushNow: false };
    site.navigationPending = null;
    site.entryPending = false;
    site.showExits = false;
    state.currentAreaId = 0;
    state.siteExplorationActive = false;
    log(`SHIP BOARD | from=${from.areaId} via=${source} node=0`);
    return true;
  }
  const firstVisit = !dest.visited;
  // Upgrade legacy unexplored nodes on entry without rerolling their contents.
  // Never assign a new room identity to a legacy area already described by the AI.
  if (firstVisit && !dest.roomScene) dest.roomScene = generateRoomScene(site);
  if (firstVisit && !dest.roomTitle) dest.roomTitle = getGraphRoomTitle(dest);
  if (firstVisit && !dest.discoveryScene) dest.discoveryScene = rollRoomDiscoveryNarrative(dest.result);
  dest.visited = true;
  dest.visitCount++;
  if (firstVisit) {
    eoeArmSiteMechanicalTrap(site,dest,true);
    logGraphAreaReveal(dest);
    activateGraphCreature(dest);
    triggerGraphEncounter(dest, source);
  }
  const ambushNow = !!exit.status.ambush && !exit.status.ambushTriggered;
  if (ambushNow) exit.status.ambushTriggered = true;
  site.currentAreaId = dest.areaId;
  state.currentAreaId = dest.areaId; // compatibility with older scripts
  state.siteExplorationCount = Object.values(site.nodes).filter(n => n.areaId !== 0 && n.visited).length;
  site.lastMove = { from: from.areaId, to: dest.areaId, exitNumber: exit.number,
    passageId: exit.passageId, connector: exit.connector, ambushNow, firstVisit,
    source, needsNarration: source !== "AI",
    // Stable across repeated OUTPUT calls for this same action. Unlike
    // site.showExits, this is not consumed during rendering.
    actionCount: typeof info !== "undefined" && Number.isFinite(info.actionCount)
      ? info.actionCount : null };
  site.navigationPending = null;
  site.showExits = true;
  log(`NAV COMMIT | via=${source} ${from.areaId}->${dest.areaId} exit=${exit.number} ` +
      `new=${firstVisit} ambush=${ambushNow} visited=${state.siteExplorationCount}/${site.totalAreas}`);
  return true;
};

/**
 * enterGraphSite(site, source)
 * The crew crosses into the Site for the first time. Start exploration in
 * the graph entry node; do not auto-complete that node's challenges.
 */
const enterGraphSite = (site, source) => {
  if (!site || site.mode !== "AT_SHIP") return false;

  if (!node) return false;

  if (firstVisit && !node.roomScene) node.roomScene = generateRoomScene(site);

  if (firstVisit) {
    eoeArmSiteMechanicalTrap(site,node,true);

  }

/**
 * graphExitDirectionCompatible(exit, input)
 * Refuse a staircase exit if the player explicitly chose the opposite vertical
 * direction. Used in INPUT scoring AND AI candidate filtering.
 */
const graphExitDirectionCompatible = (exit, input) => {
  const t = String(input || "").toLowerCase();
  const up = /\b(up|upward|upwards|ascend|ascending|higher)\b/.test(t);
  const down = /\b(down|downward|downwards|descend|descending|lower)\b/.test(t);
  const c = String(exit.connector || "").toLowerCase();
  if (up && !down && c.includes("stairs going down")) return false;
  if (down && !up && c.includes("stairs going up")) return false;
  return true;
};

/**
 * scoreGraphExitMatches(input, site)
 * Conservative keyword matching. This function does NOT decide whether
 * movement was intended; INPUT must first establish positive movement intent.
 * Score connector words and fixed landmarks, then require a clear winner.
 */
const scoreGraphExitMatches = (input, site) => {
  const normalized = ` ${String(input).toLowerCase().replace(/[^a-z0-9]+/g, " ").trim()} `;
  const common = new Set(["a", "the", "to", "of", "and", "in", "with", "from", "into", "past", "light"]);
  return getSiteExits(site).map(exit => {
    if (!graphExitDirectionCompatible(exit, input)) return { number: exit.number, destination: exit.to, score: -999 };
    let score = 0;
    const landmarkPhrase = ` ${exit.landmark.toLowerCase()} `;
    if (normalized.includes(landmarkPhrase)) score += 5;
    else for (const word of exit.landmark.toLowerCase().split(" ")) {
      if (word.length > 3 && !common.has(word) && normalized.includes(` ${word} `)) score += 3;
    }
    for (const word of graphConnectorVocabulary(exit.connector)) {
      if (word.length > 3 && normalized.includes(` ${word} `)) {
        score += ["stairs", "staircase", "bridge", "doorway", "archway"].includes(word) ? 3 : 2;
      }
    }
    // Directional words are weak alone, but break ties for stairways.
    const connector = exit.connector.toLowerCase();
    if (connector.includes("stairs going down") && /\b(down|descend|descending|lower)\b/.test(normalized)) score += 1;
    if (connector.includes("stairs going up") && /\b(up|ascend|ascending|higher)\b/.test(normalized)) score += 1;
    return { number: exit.number, destination: exit.to, score };
  }).sort((a, b) => b.score - a.score || a.number - b.number);
};

/**
 * graphInputHasMovementIntent(text)
 * Reject inquiries, intentions, inspection and hypothetical instructions.
 * Accept clear navigation verbs. False negatives are preferable to unwanted
 * room transitions; #EXPLORE<number> always remains the exact override.
 */
const graphInputHasMovementIntent = (text) => {
  const t = String(text || "").toLowerCase();
  // A direct order to a named NPC is not movement by the PLAYER.
  // Accept 1-3 word forms of address: "Suri, go...", "Captain Voss, take...".
  // The NPC may move narratively, but player location must not change.
  if (/^\s*(?:[a-z][\w'-]*\s+){0,2}[a-z][\w'-]*\s*,\s*(?:please\s+)?(?:go|walk|head|enter|step|move|climb|descend|ascend|proceed|follow|cross|pass|crawl|venture|return|backtrack|take)\b/.test(t)) return false;
  if (!t.trim() || t.includes("?") || /\b(don't|do not|won't|not going|never|without)\b/.test(t)) return false;
  if (/\b(want to|plan to|thinking of|consider|should we|could we|would we|ask|tell|discuss|wonder|look at|examine|inspect|point at)\b/.test(t)) return false;
  // Moving up TO a threshold isn't the same as moving THROUGH it.
  if (/\b(walk|head|go|move|step)\s+(?:over\s+)?(?:toward|towards|up to)\b/.test(t)) return false;
  if (/\b(approach|stand by|peer into|look through|look down|listen at)\b/.test(t) &&
      !/\b(walk|go|head|enter|step|move|climb|descend|ascend|proceed|follow|cross|pass|crawl|venture|return|backtrack)\b/.test(t)) return false;
  // "head" is also a BODY PART. A bare word match triggered route popups
  // during combat for inputs like "aim for the head". Require a directional
  // construction before treating "head" as the verb meaning "travel".
  const travelVerb = /\b(go|walk|enter|step|move|climb|descend|ascend|proceed|follow|cross|pass|crawl|venture|return|backtrack)\b/.test(t);
  const headRoute = /\b(?:i|we|you)\s+(?:(?:quickly|carefully|quietly|slowly|immediately)\s+)?head\s+(?:(?:straight|directly)\s+)?(?:to|into|through|down|up|back|out|across|along|inside|outside|north|south|east|west)\b/.test(t) ||
    /\b(?:i|we|you)\s+head\s+for\s+(?:the\s+|a\s+)?(?:stairs|staircase|hallway|corridor|bridge|path|route|exit|doorway|passage|door)\b/.test(t);
  const takeRoute = /\btake\s+(?:(?:the|a|this|that|an)\s+)?(?:\w+\s+){0,3}(?:stairs|staircase|hallway|corridor|bridge|path|route|exit|doorway|passage|steps|door)\b/.test(t);
  return travelVerb || headRoute || takeRoute;
};

/**
 * formatGraphDebug(site)
 * Compact, capped digest for playtest logs. Includes every node's discovered
 * status, its exits, and the target node IDs without exposing it in story.
 */
const formatGraphDebug = (site) => {
  const nodes = Object.values(site.nodes || {}).filter(n => n.areaId !== 0).sort((a,b) => a.areaId - b.areaId);
  const lines = [`GRAPH ${site.siteId} | ${nodes.length} nodes | ${Object.keys(site.passages || {}).length} passages`];
  for (const n of nodes.slice(0, 25)) {
    lines.push(`${String(n.areaId).padStart(2,"0")} ${n.visited ? "V" : "U"} ${n.result}: ` +
      n.exits.map((e,i) => `${i+1}->${e.to}`).join(" "));
  }
  if (nodes.length > 25) lines.push(`... ${nodes.length - 25} further nodes not printed`);
  return lines.join("\n");
};
